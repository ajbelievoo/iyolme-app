  Future<void> _fetchWalletBalance() async {
    try {
      // Refresh user details to get latest coin balance
      if (widget.isBoostMode) {
         await UserService.instance.fetchUserDetails();
      }

      if (widget.isBoostMode) {
        // Boost uses coins from User profile
        final user = SessionManager.instance.getUser();
        if (mounted) {
          setState(() {
             _coinBalance = user?.coinWallet;
          });
        }
      } else {
        final w = await _repo.fetchWalletRaw();
        if (!mounted) return;
        setState(() {
          _walletBalance = w.balance;
        });
      }
    } catch (_) {
      // ignore
    }
  }
  
  void _onBudgetChanged() {
    if (_debounceTimer?.isActive ?? false) _debounceTimer!.cancel();
    _debounceTimer = Timer(const Duration(milliseconds: 800), _fetchEstimate);
  }

  Future<void> _fetchEstimate() async {
    final daily = _dailyBudget ?? 0;
    final total = _totalBudget ?? 0;
    
    // Simple duration calc if total > daily
    int days = _durationDays;
    if (daily > 0 && total > 0) {
      days = (total / daily).ceil();
      if (days < 1) days = 1;
    }

    if (daily <= 0 || total <= 0) {
      setState(() {
        _adEstimate = null;
        _boostEstimate = null;
      });
      return;
    }

    setState(() {
      _isFetchingEstimate = true;
    });

    try {
      if (widget.isBoostMode) {
        final placementParam = _placementParam(_placement);
        final effectivePlacement =
            placementParam == 'profile' ? 'feed' : placementParam;
        final res = await BoostService.instance.getEstimate(
          budgetCoins: total.toInt(),
          durationHours: days * 24,
          placement: effectivePlacement,
        );
        if (mounted) {
          setState(() {
            // Fallback estimates if backend returns 0/0
            final data = res.data;
            final rMin = data?.estimatedReachMin ?? 0;
            final rMax = data?.estimatedReachMax ?? 0;
            final iMin = data?.estimatedImpressionsMin ?? 0;
            final iMax = data?.estimatedImpressionsMax ?? 0;
            if ((rMin <= 0 && rMax <= 0) && (iMin <= 0 && iMax <= 0)) {
              int base = (total * 100).toInt();
              switch (effectivePlacement) {
                case 'reel':
                  base = (base * 1.8).toInt();
                  break;
                case 'story':
                  base = (base * 1.4).toInt();
                  break;
                case 'feed':
                  base = (base * 1.2).toInt();
                  break;
                default:
                  break;
              }
              final impMin = base;
              final impMax = (base * 1.6).toInt();
              final reachMin = (impMin * 0.6).toInt();
              final reachMax = (impMax * 0.7).toInt();
              _boostEstimate = BoostEstimateData(
                estimatedImpressionsMin: impMin,
                estimatedImpressionsMax: impMax,
                estimatedReachMin: reachMin,
                estimatedReachMax: reachMax,
                estimatedClicksMin: null,
                estimatedClicksMax: null,
              );
            } else {
              _boostEstimate = data;
            }
            _adEstimate = null;
          });
        }
      } else {
        // Ads uses money
        final res = await UserAdsService.instance.getEstimate(
          budget: total.toDouble(),
          durationHours: days * 24,
          placement: _placementParam(_placement),
          billing: 'cpc', // Defaulting to CPC for wizard
        );
        if (mounted) {
          setState(() {
            _adEstimate = res.data;
            _boostEstimate = null;
          });
        }
      }
    } catch (e) {
      if (widget.isBoostMode) {
        final placementParam = _placementParam(_placement);
        final effectivePlacement =
            placementParam == 'profile' ? 'feed' : placementParam;
        final total = _totalBudget ?? 0;
        int base = (total * 100).toInt();
        switch (effectivePlacement) {
          case 'reel':
            base = (base * 1.8).toInt();
            break;
          case 'story':
            base = (base * 1.4).toInt();
            break;
          case 'feed':
            base = (base * 1.2).toInt();
            break;
          default:
            break;
        }
        final impMin = base;
        final impMax = (base * 1.6).toInt();
        final reachMin = (impMin * 0.6).toInt();
        final reachMax = (impMax * 0.7).toInt();
        if (mounted) {
          setState(() {
            _boostEstimate = BoostEstimateData(
              estimatedImpressionsMin: impMin,
              estimatedImpressionsMax: impMax,
              estimatedReachMin: reachMin,
              estimatedReachMax: reachMax,
              estimatedClicksMin: null,
              estimatedClicksMax: null,
            );
            _adEstimate = null;
          });
        }
      }
    } finally {
      if (mounted) {
        setState(() {
          _isFetchingEstimate = false;
        });
      }
    }
  }

  Future<void> _openPostPicker() async {
    final myId = SessionManager.instance.getUserID();
    if (myId <= 0) {
      BaseController.share.showSnackBar('Please login again');
      return;
    }

      final postsData = await PostService.instance.fetchUserPosts(
        type: PostType.posts,
        userId: myId,
        lastItemId: null,
      );
      final reelsData = await PostService.instance.fetchUserPosts(
        type: PostType.reels,
        userId: myId,
        lastItemId: null,
      );

      final list = <Post>[];
      list.addAll(postsData?.pinnedPostList ?? const <Post>[]);
      list.addAll(postsData?.posts ?? const <Post>[]);
      list.addAll(reelsData?.pinnedPostList ?? const <Post>[]);
      list.addAll(reelsData?.posts ?? const <Post>[]);

      final seen = <int>{};
      final uniq = <Post>[];
      for (final p in list) {
        final id = p.id ?? 0;
        if (id <= 0) continue;
        if (seen.contains(id)) continue;
        seen.add(id);
        uniq.add(p);
      }
      return uniq;
    }

      context: context,
      isScrollControlled: true,
      backgroundColor: scaffoldBackgroundColor(context),
      shape: const RoundedRectangleBorder(
        borderRadius: BorderRadius.vertical(top: Radius.circular(16)),
      ),
      builder: (ctx) {
        return SafeArea(
          child: Padding(
            padding: EdgeInsets.only(
              left: 16,
              right: 16,
              top: 12,
              bottom: MediaQuery.of(ctx).viewInsets.bottom + 12,

            child: SizedBox(
              height: MediaQuery.of(ctx).size.height * 0.72,

                crossAxisAlignment: CrossAxisAlignment.start,
                children: [

                    children: [

                        icon: const Icon(Icons.close),

                            child: Text(

                              style: TextStyle(color: textLightGrey(ctx)),

                                          width: 42,
                                          height: 42,
                                          decoration: BoxDecoration(
                                            color: Theme.of(ctx2).cardColor,

                                          'Post #$id',
                                          style: const TextStyle(
                                            fontWeight: FontWeight.w900,
                                          ),
                                        ),
                                        subtitle: Text(
                                          caption.isNotEmpty
                                              ? caption
                                              : (thumb.isNotEmpty ? 'Thumbnail available' : ''),
                                          maxLines: 2,
                                          overflow: TextOverflow.ellipsis,
                                          style: TextStyle(
                                            fontSize: 12,
                                            color: textLightGrey(ctx2),

                                        ),

  void _clearSelectedPost() {
    setState(() {
      _selectedPostId = null;
      _selectedPostCaption = null;
      _selectedPostType = null;
      _selectedPostPreviewUrl = null;
    });
  }

  Future<void> _pickAndUploadMedia() async {
    final media = await MediaPickerHelper.shared.pickMedia();
    if (media == null) return;

    setState(() {
      _isMediaUploading = true;
    });

    try {
      // BaseController.share.showLoader(); // Loader handled by button state now
      final res = await CommonService.instance.uploadFileGivePath(media.file);
      // BaseController.share.stopLoader();

      if (res.status != true || (res.data ?? '').toString().trim().isEmpty) {
        BaseController.share.showSnackBar(res.message ?? 'Upload failed');
        return;
      }

      final url = res.data!.toString().trim();
      if (!mounted) return;
      setState(() {
        _mediaUrl = url;
        _clearSelectedPost();
      });
      BaseController.share.showSnackBar('Media uploaded');
    } catch (e) {
      // BaseController.share.stopLoader();
      BaseController.share.showSnackBar(e.toString());
    } finally {
      if (mounted) {
        setState(() {
          _isMediaUploading = false;
        });
      }
    }
  }

    switch (p) {
      case CampaignPlacement.feed:
        return 'feed';
      case CampaignPlacement.story:
        return 'story';
      case CampaignPlacement.reel:
        return 'reel';
      case CampaignPlacement.profile:
        return 'profile';
      case CampaignPlacement.explore:
        return 'explore';
    }
  }

  @override
  void initState() {
    super.initState();

    _fetchWalletBalance();
    _loadCountries();

    _titleController.text = widget.postTitle?.trim().isNotEmpty == true
        ? widget.postTitle!.trim()
        : 'My campaign';

    if (widget.isBoostMode) {
      _goal = CampaignGoal.promotePost;
      _audienceCountry = widget.prefillAudienceCountry ?? 'all';
      _dailyBudget = widget.prefillDailyBudget ?? 5;
      _totalBudget = widget.prefillTotalBudget ?? 25;
      _mediaUrl = widget.mediaUrl;

      _selectedPostId = widget.postId;
      _selectedPostCaption = widget.postCaption;
      _selectedPostType = widget.postType;
      _selectedPostPreviewUrl = widget.mediaUrl;

      final postType = widget.postType;
      if (postType == PostType.video) {
        _placement = CampaignPlacement.reel;
      } else {
        _placement = CampaignPlacement.feed;
      }
    }

    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (!mounted) return;
      if (widget.isBoostMode && widget.openQuickBudgetOnStart) {
        _openQuickBudgetIfNeeded();
      }
    });
  }

  @override
  void dispose() {
    _titleController.dispose();
    _debounceTimer?.cancel();
    super.dispose();
  }

  void _openQuickBudgetIfNeeded() {
    if (_didOpenQuickBudget) return;
    _didOpenQuickBudget = true;

    showModalBottomSheet<void>(
      context: context,
      isScrollControlled: true,
      backgroundColor: scaffoldBackgroundColor(context),
      shape: const RoundedRectangleBorder(
        borderRadius: BorderRadius.vertical(top: Radius.circular(16)),
      ),
      builder: (_) {
        return BoostPostBottomSheet(
          postId: widget.postId,
          initialDailyBudget: (_dailyBudget ?? 0) > 0 ? _dailyBudget! : 5,
          initialTotalBudget: (_totalBudget ?? 0) > 0 ? _totalBudget! : 25,
          onBoostNow: (daily, total) {
            setState(() {
              _dailyBudget = daily;
              _totalBudget = total;
            });
            _submitBoostApi(dailyBudget: daily, totalBudget: total);
          },
        );
      },
    );
  }

  Future<void> _submitBoostApi({
    required num dailyBudget,
    required num totalBudget,
    bool forceDraft = false,
  }) async {
    final title = (widget.postTitle?.trim().isNotEmpty == true)
        ? widget.postTitle!.trim()
        : _titleController.text.trim().isNotEmpty
            ? _titleController.text.trim()
            : 'Boosted post';

    try {
      final wb = _coinBalance;
      final bool isDraft = forceDraft || (wb != null && totalBudget > wb);

      final media = (widget.mediaUrl ?? _mediaUrl)?.trim();
      final effectivePostId = _selectedPostId ?? widget.postId ?? 0;
      if ((effectivePostId == 0) && (media == null || media.isEmpty)) {
        BaseController.share.showSnackBar('Please provide media');
        return;
      }

      final int total = totalBudget.toInt();
      int days = _durationDays;
      if (dailyBudget > 0 && totalBudget > 0) {
        days = (totalBudget / dailyBudget).ceil();
        if (days < 1) days = 1;
      }

      final meta = <String, dynamic>{
        'source': 'boost',
        'isDraft': isDraft,
      };

      final res = await BoostService.instance.createBoostRequest(
        postId: (_selectedPostId ?? widget.postId ?? 0),
        budgetCoins: total,
        durationHours: days * 24,
        placement: _placementParam(_placement),
        title: title,
        targetCountry: _audienceCountry,
        metadataJson: jsonEncode(meta),
        status: isDraft ? 'draft' : null,
      );
      if (res.status != true) {
        throw Exception(res.message ?? 'Failed to submit boost');
      }

      BaseController.share.showSnackBar(isDraft ? 'Campaign saved as draft' : 'Campaign submitted');

      if (!mounted) return;

      if (!widget.redirectToAnalyticsOnSubmit) {
        Navigator.of(context).maybePop();
        return;
      }

      Navigator.of(context).pushReplacement(
        MaterialPageRoute(
          builder: (_) => const AdsManagerScreen(initialTabIndex: 1),
        ),
      );
    } catch (e) {
      BaseController.share.showSnackBar(e.toString());
    }
  }

  Future<void> _submitWizardApi() async {
    await _submitWizardApiInternal(forceDraft: false);
  }

  Future<void> _submitWizardApiInternal({required bool forceDraft}) async {
    final title = _titleController.text.trim();
    if (title.isEmpty) {
      BaseController.share.showSnackBar('Please enter campaign name');
      return;
    }
    if (_goal == null) {
      BaseController.share.showSnackBar('Please select goal');
      return;
    }
    if ((_audienceCountry ?? '').trim().isEmpty) {
      BaseController.share.showSnackBar('Please enter country');
      return;
    }
    if ((_dailyBudget ?? 0) <= 0 || (_totalBudget ?? 0) <= 0) {
      BaseController.share.showSnackBar('Please enter valid budgets');
      return;
    }

    final hasPost = (_selectedPostId ?? 0) > 0;
    final media = (_mediaUrl ?? '').trim();
    if (!hasPost && media.isEmpty) {
      BaseController.share.showSnackBar('Please provide media');
      return;
    }

    try {
      final wb = _walletBalance;
      final total = _totalBudget ?? 0;
      final bool isDraft = forceDraft || (wb != null && total > wb);
      
      final campaignId = await _repo.createCampaign(
        request: AdsManagerCampaignCreateRequestDto(
          title: title,
          goal: _goalParam(_goal!),
          placement: _placementParam(_placement),
          audienceCountry: _audienceCountry,
          dailyBudget: _dailyBudget,
          totalBudget: _totalBudget,
          mediaUrl: media.isEmpty ? null : media,
          postId: hasPost ? _selectedPostId : null,
          status: isDraft ? 'draft' : null,
          metadata: <String, dynamic>{
            'source': 'wizard',
            'isDraft': isDraft,
          },
        ),
      );

      BaseController.share.showSnackBar(isDraft ? 'Campaign saved as draft' : 'Campaign submitted');

      if (!mounted) return;

      if (!widget.redirectToAnalyticsOnSubmit) {
        Navigator.of(context).maybePop();
        return;
      }

      Navigator.of(context).pushReplacement(
        MaterialPageRoute(
          builder: (_) => const AdsManagerScreen(initialTabIndex: 1),
        ),
      );
    } catch (e) {
      final msg = e.toString().toLowerCase();
      if (msg.contains('403')) {
        final media = (_mediaUrl ?? '').trim();
        if (media.isEmpty) {
          BaseController.share.showSnackBar('Forbidden: please upload media to create user ad');
          return;
        }
        final d = _durationDays > 0 ? _durationDays : 1;
        try {
          final status = await UserAdsService.instance.createUserAdRequest(
            title: title,
            description: null,
            mediaPath: media,
            mediaType: 'image',
            placement: _placementParam(_placement),
            billing: 'cpc',
            budget: (_totalBudget ?? 0).toDouble(),
            durationHours: d * 24,
            ctaUrl: null,
            keywords: null,
            metadataJson: null,
          );
          if (status.status == true) {
            BaseController.share.showSnackBar('Ad request submitted');
            if (!mounted) return;
            Navigator.of(context).pushReplacement(
              MaterialPageRoute(
                builder: (_) => const AdsManagerScreen(initialTabIndex: 1),
              ),
            );
            return;
          }
        } catch (_) {}
      }
      BaseController.share.showSnackBar(e.toString());
    }
  }

  bool get _canContinue {
    if (_step == 0) return _titleController.text.trim().isNotEmpty;
    if (_step == 1) return _goal != null;
    if (_step == 2) return !_isMediaUploading; // Block if uploading
    if (_step == 3) return true;
    if (_step == 4) return (_audienceCountry ?? '').trim().isNotEmpty;
    if (_step == 5) return (_dailyBudget ?? 0) > 0 && (_totalBudget ?? 0) > 0;
    // Step 6 is Preview, always valid
    return true;
  }

class _PreviewStep extends StatelessWidget {
  const _PreviewStep({
    required this.goal,
    required this.placement,
    required this.country,
    required this.dailyBudget,
    required this.totalBudget,
    required this.mediaUrl,
    required this.selectedPostId,
    required this.loadingEstimate,
    this.estAd,
    this.estBoost,
  });

  final CampaignGoal? goal;
  final CampaignPlacement placement;
  final String? country;
  final num? dailyBudget;
  final num? totalBudget;
  final String? mediaUrl;
  final int? selectedPostId;
  final bool loadingEstimate;
  final AdEstimateData? estAd;
  final BoostEstimateData? estBoost;

  @override
  Widget build(BuildContext context) {
    final cc = (country ?? '').trim();
    final countryLabel = cc.isEmpty
        ? '-'
        : (cc.toLowerCase() == 'all')
            ? 'All'
            : cc.toUpperCase();
    final hasPost = (selectedPostId ?? 0) > 0;
    final media = (mediaUrl ?? '').trim();
    final mediaLabel = hasPost
        ? 'Post #$selectedPostId '
        : (media.isNotEmpty ? media : '-');

    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        const SizedBox(height: 20),
        const AdsSectionHeader(title: 'Estimated Results'),
        const SizedBox(height: 10),
        if (loadingEstimate)
          Container(
            padding: const EdgeInsets.symmetric(vertical: 24, horizontal: 16),
            width: double.infinity,
            decoration: BoxDecoration(
              color: Theme.of(context).cardColor,
              borderRadius: BorderRadius.circular(16),
              border: Border.all(
                color: textLightGrey(context).withValues(alpha: 0.16),
              ),
            ),
            child: Column(
              children: [
                const SizedBox(
                  width: 24,
                  height: 24,
                  child: CircularProgressIndicator(strokeWidth: 2),
                ),
                const SizedBox(height: 12),
                Text(
                  'Calculating estimated reach...',
                  style: TextStyle(
                    color: textLightGrey(context),
                    fontWeight: FontWeight.w600,
                  ),
                ),
              ],
            ),
          )
        else if (estAd != null || estBoost != null)
          Container(
            padding: const EdgeInsets.all(16),
            width: double.infinity,
            decoration: BoxDecoration(
              color: Colors.green.withValues(alpha: 0.08),
              borderRadius: BorderRadius.circular(16),
              border: Border.all(
                color: Colors.green.withValues(alpha: 0.3),
              ),
            ),
            child: Column(
              children: [
                if (estAd != null) ...[
                  _estRow(context, 'Impressions',
                      '${estAd!.estimatedImpressionsMin} - ${estAd!.estimatedImpressionsMax}'),
                  const SizedBox(height: 8),
                  _estRow(context, 'Clicks',
                      '${estAd!.estimatedClicksMin ?? 0} - ${estAd!.estimatedClicksMax ?? 0}'),
                  const SizedBox(height: 8),
                  _estRow(context, 'Reach',
                      '${(estAd!.estimatedImpressionsMin ?? 0) * 0.8} - ${(estAd!.estimatedImpressionsMax ?? 0) * 0.9}'),
                ],
                if (estBoost != null) ...[
                   _estRow(context, 'Reach',
                      '${estBoost!.estimatedReachMin} - ${estBoost!.estimatedReachMax}'),
                  const SizedBox(height: 8),
                  _estRow(context, 'Impressions',
                      '${estBoost!.estimatedImpressionsMin} - ${estBoost!.estimatedImpressionsMax}'),
                ],
              ],
            ),
          )
        else
          Container(
            padding: const EdgeInsets.all(16),
            width: double.infinity,
            decoration: BoxDecoration(
              color: Theme.of(context).cardColor,
              borderRadius: BorderRadius.circular(16),
              border: Border.all(
                color: textLightGrey(context).withValues(alpha: 0.16),
              ),
            ),
            child: Center(
              child: Text(
                'Enter budget to see estimates',
                style: TextStyle(color: textLightGrey(context)),
              ),
            ),
          ),
        
        const SizedBox(height: 20),
        const AdsSectionHeader(title: 'Campaign Summary'),
        const SizedBox(height: 10),
        Container(
          padding: const EdgeInsets.all(12),
          decoration: BoxDecoration(
            color: Theme.of(context).cardColor,
            borderRadius: BorderRadius.circular(16),
            border: Border.all(
              color: textLightGrey(context).withValues(alpha: 0.16),
            ),
          ),
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              _summaryRow(context, 'Goal', goal?.name ?? '-'),
              const Divider(height: 16),
              _summaryRow(context, 'Placement', placement.name),
              const Divider(height: 16),
              _summaryRow(context, 'Audience', countryLabel),
              const Divider(height: 16),
              _summaryRow(context, 'Daily Budget', '${dailyBudget ?? '-'}'),
              const Divider(height: 16),
              _summaryRow(context, 'Total Budget', '${totalBudget ?? '-'}'),
              const Divider(height: 16),
              _summaryRow(context, 'Media', mediaLabel),
            ],
          ),
        ),
      ],
    );
  }

  Widget _estRow(BuildContext context, String label, String value) {
    return Row(
      mainAxisAlignment: MainAxisAlignment.spaceBetween,
      children: [
        Text(label, style: TextStyle(color: textDarkGrey(context))),
        Text(
          value,
          style: TextStyle(
            color: textDarkGrey(context),
            fontWeight: FontWeight.w900,
          ),
        ),
      ],
    );
  }

  Widget _summaryRow(BuildContext context, String label, String value) {
    return Row(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        SizedBox(
          width: 100,
          child: Text(
            label,
            style: TextStyle(
              color: textLightGrey(context),
              fontSize: 13,
            ),
          ),
        ),
        Expanded(
          child: Text(
            value,
            style: TextStyle(
              color: textDarkGrey(context),
              fontWeight: FontWeight.w700,
              fontSize: 13,
            ),
          ),
        ),
      ],
    );
  }
}

class _PaySubmitStep extends StatelessWidget {

  final num? totalBudget;

    final total = totalBudget ?? 0;

      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        if (showWarning)
          Container(
            width: double.infinity,
            padding: const EdgeInsets.all(12),

                color: Colors.orange.withValues(alpha: 0.22),
              ),
            ),
            child: Text(
              'Insufficient balance. Available: ${wb.toStringAsFixed(2) ?? '-'} | Required: ${total.toStringAsFixed(2)}',

                fontSize: 12,
                height: 1.35,
                color: textDarkGrey(context),
                fontWeight: FontWeight.w700,
              ),

          padding: const EdgeInsets.all(16),

             color: Theme.of(context).cardColor,
             borderRadius: BorderRadius.circular(16),
             border: Border.all(color: textLightGrey(context).withValues(alpha: 0.16)),
          ),
          child: Column(
            children: [
              Row(
                mainAxisAlignment: MainAxisAlignment.spaceBetween,
                children: [
                  Text('Total Budget', style: TextStyle(color: textLightGrey(context))),
                  Text(total.toStringAsFixed(2), style: TextStyle(fontWeight: FontWeight.w900, color: textDarkGrey(context))),
                ],
              ),
              const SizedBox(height: 10),

                mainAxisAlignment: MainAxisAlignment.spaceBetween,
                children: [
                  Text('Current Balance', style: TextStyle(color: textLightGrey(context))),

              ),
              const Divider(height: 24),
              Row(
                mainAxisAlignment: MainAxisAlignment.spaceBetween,
                children: [
                  Text('Remaining after', style: TextStyle(color: textLightGrey(context))),
                  Text(((wb ?? 0) - total).toStringAsFixed(2),

                      fontWeight: FontWeight.w900,

              ),

          ),

              shape: RoundedRectangleBorder(
                borderRadius: BorderRadius.circular(14),
              ),
            ),
          ),
        ),
        const SizedBox(height: 10),
        SizedBox(
          height: 52,
          width: double.infinity,
          child: ElevatedButton.icon(
            onPressed: onSubmit,
            icon: const Icon(Icons.check_circle_outline),
            label: const Text(
              'Confirm & Submit',
              style: TextStyle(fontWeight: FontWeight.w900),
            ),
            style: ElevatedButton.styleFrom(
              shape: RoundedRectangleBorder(
                borderRadius: BorderRadius.circular(14),
              ),
            ),
          ),
        ),
      ],
    );
  }
