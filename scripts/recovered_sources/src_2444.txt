/// Phone number information from a barcode.
class Phone {
  /// Construct a new [Phone] instance.
  const Phone({

  });

  /// Create a [Phone] from the given [data].
  factory Phone.fromNative(Map<Object?, Object?> data) {

  final String? number;

  final PhoneType type;
