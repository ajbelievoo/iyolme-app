class EarningsLedgerItem {
  EarningsLedgerItem({
    this.id,
    this.type,
    this.createdAt,
    this.amountTokens,
    this.amountCoins,
    this.amountUsd,
    this.meta,
  });

  String? id;
  String? type;
  DateTime? createdAt;
  num? amountTokens;
  num? amountCoins;
  num? amountUsd;
  Map<String, dynamic>? meta;

  factory EarningsLedgerItem.fromJson(Map<String, dynamic> json) {
    num? readNum(String key) {
      final v = json[key];
      if (v is num) return v;
      return num.tryParse('${v ?? ''}'.trim());
    }

    final createdRaw = json['created_at']?.toString();
    final created = createdRaw == null ? null : DateTime.tryParse(createdRaw);

    final rawMeta = json['meta'];
    Map<String, dynamic>? meta;
    if (rawMeta is Map<String, dynamic>) {
      meta = rawMeta;
    } else if (rawMeta is Map) {
      meta = rawMeta.cast<String, dynamic>();
    }

    return EarningsLedgerItem(
      id: json['id']?.toString(),
      type: json['type']?.toString(),
      createdAt: created,
      amountTokens: readNum('amount_tokens'),
      amountCoins: readNum('amount_coins'),
      amountUsd: readNum('amount_usd'),
      meta: meta,
    );
  }
}
