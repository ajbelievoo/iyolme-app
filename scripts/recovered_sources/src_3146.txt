class _FullNameWithBlueTickState extends State<FullNameWithBlueTick> {
  static final RxMap<int, int> _verifiedCache = <int, int>{}.obs;
  static final Map<int, DateTime> _lastResolveAttemptAt = <int, DateTime>{};
  static const Duration _resolveThrottle = Duration(seconds: 20);
  final RxInt _resolvedVerify = 0.obs;
  final RxBool _isResolving = false.obs;

  bool _isVerified(dynamic v) {
    if (v == null) return false;
    if (v is bool) return v;
    if (v is num) return v.toInt() == 1;
    if (v is String) {
      final s = v.trim().toLowerCase();
      if (s == 'true' || s == 'yes') return true;
      final n = int.tryParse(s);
      return n == 1;
    }
    return false;
  }

  @override
  void initState() {
    super.initState();
    _resolvedVerify.value = _isVerified(widget.isVerify) ? 1 : 0;
    _resolveIfNeeded();
  }

  @override
  void didUpdateWidget(covariant FullNameWithBlueTick oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (oldWidget.userId != widget.userId || oldWidget.isVerify != widget.isVerify) {
      _resolvedVerify.value = _isVerified(widget.isVerify) ? 1 : 0;
      _resolveIfNeeded();
    }
  }

  Future<void> _resolveIfNeeded() async {
    final uid = widget.userId;
    if (uid == null || uid <= 0) return;

    final last = _lastResolveAttemptAt[uid];
    if (last != null && DateTime.now().difference(last) < _resolveThrottle) {
      return;
    }

    if (_resolvedVerify.value == 1) {
      _verifiedCache[uid] = 1;
      return;
    }
    final cached = _verifiedCache[uid];
    if (cached != null) {
      _resolvedVerify.value = cached;
      return;
    }
    if (_isResolving.value) return;
    _isResolving.value = true;
    _lastResolveAttemptAt[uid] = DateTime.now();
    try {
      final user = await UserService.instance.fetchUserDetails(userId: uid);
      final v = (user?.isVerify ?? 0).toInt();
      _verifiedCache[uid] = v;
      _resolvedVerify.value = v;
    } catch (_) {
      // Don't cache failures as 0; allow retry later.
    } finally {
      _isResolving.value = false;
    }
  }
