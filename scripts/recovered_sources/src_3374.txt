  Future<String?> _resolveLutFilePathForLivekit() async {
    try {
      if (arFilterPreset.value != 'lut') return null;
      final input = arFilterLutAssetPath.value.trim();
      if (input.isEmpty) return null;

      final cached = _lutResolvedPathCache[input];
      if (cached != null && cached.trim().isNotEmpty) {
        return cached;
      }

      if (input.startsWith('http://') || input.startsWith('https://')) {
        if (_lutDownloadInFlight.contains(input)) {
          return null;
        }
        _lutDownloadInFlight.add(input);
        // Non-blocking download: avoid freezing UI thread.
        DefaultCacheManager().getSingleFile(input).then((f) {
          try {
            final p = f.path;
            if (p.trim().isNotEmpty) {
              _lutResolvedPathCache[input] = p;
              // If user is still on LUT preset, re-apply to push file path to native.
              if (arFilterPreset.value == 'lut') {
                applyLivekitArSettings();
              }
            }
          } catch (_) {}
        }).whenComplete(() {
          _lutDownloadInFlight.remove(input);
        });
        return null;
      }
      return null;
    } catch (e) {
      Loggers.error('[LIVEKIT] resolve lut file path failed: $e');
      return null;
    }
  }

  Future<void> toggleLiveMusicRepeat() async {
    if (!canControlLiveMusic) return;
    final st = Map<String, dynamic>.from(liveMusicState.value);
    if (st.isEmpty || st['enabled'] != true) return;
    final now = DateTime.now().millisecondsSinceEpoch;
    st['repeat'] = !(st['repeat'] == true);
    st['updatedAtMs'] = now;
    st['issuedBy'] = myUserId;
    await updateLiveStreamData(liveMusicState: st);
  }

  Future<void> playLiveMusicQueueIndex(int index) async {
    if (!canControlLiveMusic) return;
    final st = Map<String, dynamic>.from(liveMusicState.value);
    if (st.isEmpty || st['enabled'] != true) return;

    final q = liveMusicQueue;
    if (q.isEmpty) return;
    final idx = index.clamp(0, q.length - 1);

    final item = q[idx];
    final now = DateTime.now().millisecondsSinceEpoch;
    st['queueIndex'] = idx;
    st['musicId'] = item['musicId'];
    st['title'] = item['title'];
    st['artist'] = item['artist'];
    st['image'] = item['image'];
    st['soundUrl'] = item['soundUrl'];
    st['playing'] = true;
    st['startedAtMs'] = now;
    st['startPositionMs'] = 0;
    st['pausedPositionMs'] = 0;
    st['updatedAtMs'] = now;
    st['issuedBy'] = myUserId;
    await updateLiveStreamData(liveMusicState: st);
  }

  Future<void> seekLiveMusicTo(int positionMs) async {
    if (!canControlLiveMusic) return;
    final st = Map<String, dynamic>.from(liveMusicState.value);
    if (st.isEmpty || st['enabled'] != true) return;

    final now = DateTime.now().millisecondsSinceEpoch;
    final playing = st['playing'] == true;
    final safe = positionMs < 0 ? 0 : positionMs;

    if (playing) {
      st['startedAtMs'] = now;
      st['startPositionMs'] = safe;
      st['pausedPositionMs'] = safe;
    } else {
      st['pausedPositionMs'] = safe;
      st['startPositionMs'] = safe;
    }
    st['updatedAtMs'] = now;
    st['issuedBy'] = myUserId;

    // Host should feel immediate.
    if (_useNativeLiveMusicMixing && isHost) {
      try {
        await webrtc_audio.NativeAudioManagement.liveMusicMixSeek(safe);
      } catch (_) {}
    }

    await updateLiveStreamData(liveMusicState: st);
  }

  void _startLiveMusicProgressPolling() {
    _liveMusicProgressTimer?.cancel();
    _liveMusicProgressTimer = Timer.periodic(
      const Duration(seconds: 1),
      (_) async {
        try {
          if (!isLiveMusicEnabled) {
            liveMusicPositionMs.value = 0;
            liveMusicDurationMs.value = 0;
            return;
          }

          // Always drive UI from just_audio monitor playback.
          // Native mixing position/duration can be 0/unstable on some devices.
          liveMusicPositionMs.value = liveMusicPlayer.position.inMilliseconds;
          final d = liveMusicPlayer.duration;
          liveMusicDurationMs.value = d?.inMilliseconds ?? 0;

          // Optional: keep native duration in logs for debugging.
          if (_useNativeLiveMusicMixing && isHost) {
            try {
              final dur = await webrtc_audio.NativeAudioManagement
                  .liveMusicMixGetDurationMs();
              if (dur <= 0 && (d?.inMilliseconds ?? 0) > 0) {
                // ok (fallback used)
              }
            } catch (e, st) {
              Loggers.error('[LIVE_MUSIC] native duration read failed: $e\n$st');
            }
          }
        } catch (e, st) {
          Loggers.error('[LIVE_MUSIC] progress poll failed: $e\n$st');
        }
      },
    );
  }

  void _stopLiveMusicProgressPolling() {
    _liveMusicProgressTimer?.cancel();
    _liveMusicProgressTimer = null;
  }

  StreamSubscription<dynamic>? _liveMusicPlayerStateSub;

  Future<void> _applyLiveMusicState(Map<String, dynamic> st) async {
    try {
      final enabled = st['enabled'] == true;
      if (!enabled) {
        _stopLiveMusicProgressPolling();
        if (_useNativeLiveMusicMixing && isHost) {
          await webrtc_audio.NativeAudioManagement.liveMusicMixStop();
          // Stop forcing mic publish when music disabled.
          await applyLocalAudioGate();
        }
        // Stop local monitor playback (audience + host monitor).
        try {
          await liveMusicPlayer.stop();
        } catch (e, st) {
          Loggers.error('[LIVE_MUSIC] stop player failed: $e\n$st');
        }
        return;
      }
      final url = (st['soundUrl'] ?? '').toString();
      if (url.trim().isEmpty) return;

      // Ensure audio output is allowed for music playback.
      try {
        final session = await AudioSession.instance;
        await session.configure(const AudioSessionConfiguration.music());
        await session.setActive(true);
      } catch (e, st) {
        Loggers.error('[LIVE_MUSIC] AudioSession activate failed: $e\n$st');
      }

      final startedAtMs = (st['startedAtMs'] is int)
          ? (st['startedAtMs'] as int)
          : int.tryParse('${st['startedAtMs'] ?? 0}') ?? 0;
      final startPositionMs = (st['startPositionMs'] is int)
          ? (st['startPositionMs'] as int)
          : int.tryParse('${st['startPositionMs'] ?? 0}') ?? 0;
      final pausedPositionMs = (st['pausedPositionMs'] is int)
          ? (st['pausedPositionMs'] as int)
          : int.tryParse('${st['pausedPositionMs'] ?? 0}') ?? 0;
      final shouldPlay = st['playing'] == true;

      _startLiveMusicProgressPolling();

      if (_useNativeLiveMusicMixing && isHost) {
        // Host: native mixing injects into published audio, but we still keep a
        // local monitor playback so host can hear the music too.
        try {
          final needLoadLocal = (_liveMusicLocalMonitorUrl ?? '') != url;
          if (needLoadLocal) {
            _liveMusicLocalMonitorUrl = url;
            await liveMusicPlayer.setUrl(url);
          }
        } catch (e, st) {
          Loggers.error('[LIVE_MUSIC] host monitor setUrl failed: $e\n$st');
        }

        final needLoad = (_liveMusicCurrentUrl ?? '') != url;
        if (needLoad) {
          _liveMusicCurrentUrl = url;
          final file = await DefaultCacheManager().getSingleFile(url);
          _liveMusicCurrentFilePath = file.path;
          await webrtc_audio.NativeAudioManagement.liveMusicMixStart(
            filePath: _liveMusicCurrentFilePath ?? '',
            volume: 0.6,
          );
        }

        if (shouldPlay) {
          final now = DateTime.now().millisecondsSinceEpoch;
          final pos = (startedAtMs > 0)
              ? (now - startedAtMs + startPositionMs)
              : startPositionMs;
          await webrtc_audio.NativeAudioManagement.liveMusicMixSeek(
              pos < 0 ? 0 : pos);
          await webrtc_audio.NativeAudioManagement.liveMusicMixResume();

          // local monitor
          try {
            final safe = Duration(milliseconds: pos.clamp(0, 1 << 31));
            await liveMusicPlayer.seek(safe);
            await liveMusicPlayer.play();
          } catch (e, st) {
            Loggers.error('[LIVE_MUSIC] host monitor play failed: $e\n$st');
          }
        } else {
          await webrtc_audio.NativeAudioManagement.liveMusicMixSeek(
              pausedPositionMs < 0 ? 0 : pausedPositionMs);
          await webrtc_audio.NativeAudioManagement.liveMusicMixPause();

          // local monitor
          try {
            final safe = Duration(
                milliseconds: pausedPositionMs.clamp(0, 1 << 31));
            await liveMusicPlayer.seek(safe);
            await liveMusicPlayer.pause();
          } catch (e, st) {
            Loggers.error('[LIVE_MUSIC] host monitor pause failed: $e\n$st');
          }
        }

        // Ensure mic publish state is re-applied so audience can hear injected music.
        await applyLocalAudioGate();
        return;
      }

      final needLoad = (_liveMusicCurrentUrl ?? '') != url;
      if (needLoad) {
        _liveMusicCurrentUrl = url;
        try {
          await liveMusicPlayer.setUrl(url);
        } catch (e, st) {
          Loggers.error('[LIVE_MUSIC] audience setUrl failed: $e\n$st');
          return;
        }
      }

      if (shouldPlay) {
        final now = DateTime.now().millisecondsSinceEpoch;
        final pos = (startedAtMs > 0)
            ? (now - startedAtMs + startPositionMs)
            : startPositionMs;
        final safe = Duration(milliseconds: pos.clamp(0, 1 << 31));
        try {
          await liveMusicPlayer.seek(safe);
          await liveMusicPlayer.play();
        } catch (e, st) {
          Loggers.error('[LIVE_MUSIC] audience play failed: $e\n$st');
        }
      } else {
        final safe = Duration(milliseconds: pausedPositionMs.clamp(0, 1 << 31));
        try {
          await liveMusicPlayer.seek(safe);
          await liveMusicPlayer.pause();
        } catch (e, st) {
          Loggers.error('[LIVE_MUSIC] audience pause failed: $e\n$st');
        }
      }
    } catch (e, st) {
      Loggers.error('[LIVE_MUSIC] apply failed: $e\n$st');
    }
  }

      final rawAllowed = data[FirebaseConst.liveMusicControlCohosts];
      final allowed = <int>[];
      if (rawAllowed is List) {
        for (final v in rawAllowed) {
          if (v is int) {

            if (parsed != null) allowed.add(parsed);

      await _applyLiveMusicState(st);
    } catch (_) {}
  }

  Future<void> playLiveMusic(Music music) async {
    if (!canControlLiveMusic) return;
    final m = music;
    final soundUrl = (m.sound?.addBaseURL() ?? '').toString();
    if (soundUrl.trim().isEmpty) {
      showSnackBar('Invalid music');
      return;
    }

    final now = DateTime.now().millisecondsSinceEpoch;
    final st = Map<String, dynamic>.from(liveMusicState.value);
    final queueItem = _musicToQueueItem(m);

    // If already enabled, treat selection as "play now" + append to queue.
    if (st.isNotEmpty && st['enabled'] == true) {
      final q = liveMusicQueue.toList(growable: true);
      q.add(queueItem);
      final idx = q.length - 1;

      st['enabled'] = true;
      st['queue'] = q;
      st['queueIndex'] = idx;
      st['repeat'] = st['repeat'] == true;

      st['musicId'] = queueItem['musicId'];
      st['title'] = queueItem['title'];
      st['artist'] = queueItem['artist'];
      st['image'] = queueItem['image'];
      st['soundUrl'] = queueItem['soundUrl'];

      st['playing'] = true;
      st['startedAtMs'] = now;
      st['startPositionMs'] = 0;
      st['pausedPositionMs'] = 0;

      st['issuedBy'] = myUserId;
      st['updatedAtMs'] = now;
      st['lastSoundUrl'] = queueItem['soundUrl'];
      await updateLiveStreamData(liveMusicState: st);
      return;
    }

    // First play initializes queue.
    final payload = <String, dynamic>{
      'enabled': true,
      'playing': true,
      'musicId': m.id,
      'title': m.title,
      'artist': m.artist,
      'image': m.image,
      'soundUrl': soundUrl,
      'queue': <Map<String, dynamic>>[queueItem],
      'queueIndex': 0,
      'repeat': false,
      'startedAtMs': now,
      'startPositionMs': 0,
      'pausedPositionMs': 0,
      'issuedBy': myUserId,
      'updatedAtMs': now,
      'lastSoundUrl': soundUrl,
    };

    await updateLiveStreamData(liveMusicState: payload);
  }

  Future<void> toggleLiveMusicPlayPause() async {
    if (!canControlLiveMusic) return;
    final st = Map<String, dynamic>.from(liveMusicState.value);
    if (st.isEmpty || st['enabled'] != true) return;

    final now = DateTime.now().millisecondsSinceEpoch;
    final playing = st['playing'] == true;

    int readInt(dynamic v) {
      if (v is int) return v;
      return int.tryParse('$v') ?? 0;
    }

    int computeNowPositionMs(Map<String, dynamic> s) {
      final startedAtMs = readInt(s['startedAtMs']);
      final startPositionMs = readInt(s['startPositionMs']);
      if (startedAtMs <= 0) return startPositionMs;
      final pos = now - startedAtMs + startPositionMs;
      return pos < 0 ? 0 : pos;
    }

    if (playing) {
      int pos;
      if (_useNativeLiveMusicMixing && isHost) {
        try {
          pos = await webrtc_audio.NativeAudioManagement
              .liveMusicMixGetPositionMs();
        } catch (_) {
          pos = computeNowPositionMs(st);
        }
      } else {
        pos = liveMusicPlayer.position.inMilliseconds;
      }
      st['playing'] = false;
      st['pausedPositionMs'] = pos;
      st['updatedAtMs'] = now;
      st['issuedBy'] = myUserId;
    } else {
      final pausedMs = readInt(st['pausedPositionMs']);
      st['playing'] = true;
      st['startedAtMs'] = now;
      st['startPositionMs'] = pausedMs;
      st['updatedAtMs'] = now;
      st['issuedBy'] = myUserId;
    }

    await updateLiveStreamData(liveMusicState: st);
  }

    await updateLiveStreamData(liveMusicState: <String, dynamic>{
      'enabled': false,

    });
  }

  Future<void> _unzipToDir(File zipFile, Directory targetDir) async {
    final bytes = await zipFile.readAsBytes();
    final archive = ZipDecoder().decodeBytes(bytes, verify: true);
    for (final entry in archive) {
      final name = entry.name;
      if (name.isEmpty) continue;
      if (p.isAbsolute(name)) continue;
      final normalized = p.normalize(name);
      if (normalized.startsWith('..')) continue;
      final outPath = p.join(targetDir.path, normalized);
      if (entry.isFile) {
        final outFile = File(outPath);
        await outFile.parent.create(recursive: true);
        final data = entry.content as List<int>;
        await outFile.writeAsBytes(data, flush: true);
      } else {
        await Directory(outPath).create(recursive: true);
      }
    }
  }

  Future<String?> _resolveDeepArEffectPath(DeepARFilters effect) async {
    final url = (effect.filterFile?.trim() ?? '').isNotEmpty
        ? effect.filterFile!.trim().addBaseURL()
        : '';
    if (url.isEmpty) return null;
    try {
      final downloadedFile = await DefaultCacheManager()
          .getSingleFile(url)
          .timeout(const Duration(seconds: 15));
      final filePath = downloadedFile.path;
      final lower = filePath.toLowerCase();
      if (!lower.endsWith('.zip')) return filePath;

      final base = await _getDeepArEffectsBaseDir();
      final safeName =
          '${effect.id}_${p.basenameWithoutExtension(filePath)}'.replaceAll(
        RegExp(r'[^a-zA-Z0-9_\\-\\.]'),
        '_',
      );
      final targetDir = Directory(p.join(base.path, safeName));
      if (await targetDir.exists()) {
        final config = File(p.join(targetDir.path, 'config.json'));
        if (await config.exists()) {
          return targetDir.path;
        }
      }
      try {
        if (await targetDir.exists()) {
          await targetDir.delete(recursive: true);
        }
        await targetDir.create(recursive: true);
        await _unzipToDir(downloadedFile, targetDir);
        return targetDir.path;
      } catch (e) {
        Loggers.error('[DeepAR] unzip failed: $e');
        return filePath;
      }
    } catch (e) {
      Loggers.error('[DeepAR] download failed: $e');
      return null;
    }
  }

  Future<void> applyLivekitArSettings() {
    _applyLivekitArDebounce?.cancel();
    final prev = _applyLivekitArCompleter;
    if (prev != null && !prev.isCompleted) {
      prev.complete();
    }
    final c = Completer<void>();
    _applyLivekitArCompleter = c;
    _applyLivekitArDebounce =
        Timer(const Duration(milliseconds: 120), () async {
      try {
        await _applyLivekitArSettingsNow();
      } finally {
        if (!c.isCompleted) c.complete();
        if (identical(_applyLivekitArCompleter, c)) {
          _applyLivekitArCompleter = null;
        }
      }
    });
    return c.future;
  }

  String _stableJsonEncode(Object? value) {
    Object? normalize(Object? v) {
      if (v is num) {
        return double.parse(v.toDouble().toStringAsFixed(4));
      }
      if (v is Map) {
        final entries = v.entries
            .map((e) => MapEntry(e.key.toString(), normalize(e.value)))
            .toList()
          ..sort((a, b) => a.key.compareTo(b.key));
        return <String, Object?>{for (final e in entries) e.key: e.value};
      }
      if (v is Iterable) {
        return v.map(normalize).toList(growable: false);
      }
      return v;
    }

    return jsonEncode(normalize(value));
  }

  @override
  void onInit() {
    super.onInit();

    // Live music: auto-next / repeat when track completes.
    // Only host is allowed to advance the shared state.
    try {
      _liveMusicPlayerStateSub?.cancel();
    } catch (_) {}
    _liveMusicPlayerStateSub = liveMusicPlayer.playerStateStream.listen(
      (ps) async {
        try {
          if (!isHost) return;
          if (!isLiveMusicEnabled) return;
          if (!isLiveMusicPlaying) return;
          if (ps.processingState != ProcessingState.completed) return;

          final now = DateTime.now().millisecondsSinceEpoch;
          if (now - _lastLiveMusicCompleteHandledMs < 1200) return;
          _lastLiveMusicCompleteHandledMs = now;

          if (isLiveMusicRepeat) {
            final st = Map<String, dynamic>.from(liveMusicState.value);
            if (st.isEmpty || st['enabled'] != true) return;
            st['playing'] = true;
            st['startedAtMs'] = now;
            st['startPositionMs'] = 0;
            st['pausedPositionMs'] = 0;
            st['updatedAtMs'] = now;
            st['issuedBy'] = myUserId;
            await updateLiveStreamData(liveMusicState: st);
          } else {
            await nextLiveMusic();
          }
        } catch (e, st) {
          Loggers.error('[LIVE_MUSIC] completion handler failed: $e\n$st');
        }
      },
      onError: (e) {
        Loggers.error('[LIVE_MUSIC] playerStateStream error: $e');
      },
    );

    if (_bootstrapProvider) {
      _bootstrapProviderAndInit();
    } else {
      _ensureLiveKitUserState();
      listenLiveStreamData();
      listenUserState();
      fetchLiveStreamComments();
      WakelockPlus.enable();
    }
  }

  Future<void> _applyRemoteAudioGateForUser(
    int userId,
    VideoAudioStatus status,
  ) async {
    try {
      if (userId <= 0) return;
      // Don't mute my own playback; local publishing is handled via applyLocalAudioGate.
      if (userId == myUserId) return;

      // In audio room, user should be audible only when seated AND mic status is ON.
      final shouldMute = status != VideoAudioStatus.on ||
          (isAudioRoom && !isUserSeated(userId));

      // LiveKit remote track muting handled inside LiveKit screens.
      // Keep this method for seat-map gating parity.
      if (shouldMute) {
        // no-op
      }
    } catch (_) {}
  }

  Future<void> _syncRemoteAudioGates() async {
    try {
      if (!isAudioRoom) return;
      // Apply for all known users so seat-map changes immediately stop audio.
      for (final s in liveUsersStates) {
        await _applyRemoteAudioGateForUser(s.userId, s.audioStatus);
      }
    } catch (_) {}
  }

  Future<void> lockAudioSeatAndClear(int seatIndex) async {
    final myState =
        liveUsersStates.firstWhereOrNull((e) => e.userId == myUserId);
    final isPrivileged = isHost || myState?.type == LivestreamUserType.coHost;
    if (!isPrivileged) return;
    if (!isAudioRoom) return;
    final count = liveData.value.audioSeatCount.clamp(1, 10);
    if (seatIndex < 0 || seatIndex >= count) return;

    final currentMap = Map<String, dynamic>.from(liveData.value.audioSeatMap);
    final existing = currentMap['$seatIndex'];
    final existingId = (existing is int) ? existing : int.tryParse('$existing');
    if (existingId != null && existingId > 0) {
      // Kick occupant from seat first.
      currentMap.remove('$seatIndex');
    }
    currentMap['$seatIndex'] = lockedSeatValue;
    await updateLiveStreamData(audioSeatMap: currentMap);

    if (existingId != null && existingId == myUserId) {
      await applyLocalAudioGate();
    }
  }

  Future<void> _forceExitBecauseKicked() async {
    if (_isExiting) return;

  Future<void> exitLiveOnly() async {
    _isExiting = true;
    try {
      // If host/cohost is leaving, stop live music so it doesn't continue for audience.
      try {
        final myState = liveUsersStates.firstWhereOrNull((e) => e.userId == myUserId);
        final isPrivileged = isHost || myState?.type == LivestreamUserType.coHost;
        if (isPrivileged && isLiveMusicEnabled) {
          await stopLiveMusic();
        }
      } catch (_) {}

      // Mark state as LEFT for realtime UI on other devices
      updateUserStateToFirestore(
        myUserId,
        type: LivestreamUserType.left,
        audioStatus: VideoAudioStatus.offByMe,
        videoStatus: VideoAudioStatus.offByMe,
      );

      // If cohost exits, do NOT remove cohost role.
      // Cohost rights stay until host removes/kicks or live ends.
      if (liveData.value.coHostIds?.contains(myUserId) ?? false) {
        try {
          // Stop local publishing if any (do not touch Firestore coHostIds).
          await livekitSetCameraEnabled?.call(false);
          await livekitSetMicrophoneEnabled?.call(false);
        } catch (_) {}
      }

      // Clear seat if seated
      await clearSeatForUserLatest(myUserId);

      // Disconnect without deleting livestream doc
      await livekitDisconnect?.call();
    } catch (_) {}

    // Stop listeners to avoid re-assigning seats during teardown
    try {
      liveStreamDocListener?.cancel();
      liveStreamUserStatesListener?.cancel();
      liveStreamCommentsListener?.cancel();
    } catch (_) {}

    try {
      // Host screen uses PopScope(canPop:false). Temporarily allow pop for Exit.
      allowRoutePop.value = true;
      Get.back();
    } catch (_) {}
  }

  }

  Future<void> _ensureLiveKitUserState() async {
    try {
      final user = SessionManager.instance.getUser();
      if (user == null) return;

      final userId = user.id ?? -1;
      if (userId <= 0) return;

      final userRef = liveStreamUsersRef.doc('$userId');
      if (!(await userRef.get()).exists) {
        await userRef.set(user.appUser.toJson());
      }

      final stateRef = liveStreamUserStatesRef.doc('$userId');
      final snap = await stateRef
          .withConverter(
            fromFirestore: (snapshot, _) =>
                LivestreamUserState.fromJson(snapshot.data()!),
            toFirestore: (value, _) => value.toJson(),
          )
          .get();

      final isCohostByStream =
          (liveData.value.coHostIds ?? const []).contains(userId);
      final targetType = isHost
          ? LivestreamUserType.host
          : (isCohostByStream
              ? LivestreamUserType.coHost
              : LivestreamUserType.audience);
      if (!snap.exists) {
        final initialState = user.streamState(
            time: DateTime.now().millisecondsSinceEpoch, stateType: targetType);
        await stateRef.set(initialState.toJson());
        if (!isHost) {
          _sendCommentToFirestore(type: LivestreamCommentType.joined);
        }
      } else {
        final existing = snap.data();
        await updateUserStateToFirestore(
          userId,
          type: targetType,
          audioStatus: existing?.audioStatus,
          videoStatus: existing?.videoStatus,
        );
      }
    } catch (e) {
      Loggers.error('[LIVEKIT] ensure user state failed: $e');
    }
  }

  Future<void> _startLiveKit() async {
    final wsUrl = (setting?.livekitWsUrl ?? '').trim();
    if (wsUrl.isEmpty) {
      showSnackBar('LiveKit is selected but not configured');
      return;
    }

    final user = SessionManager.instance.getUser();
    final userId = user?.id ?? -1;
    if (user == null || userId <= 0) {
      showSnackBar('User not found');
      return;
    }

    final roomName =
        (liveData.value.roomID ?? liveData.value.hostId?.toString() ?? '')
            .trim();
    if (roomName.isEmpty) {
      showSnackBar('Invalid live room');
      return;
    }

    showLoader();
    try {
      final tokenResp = await LiveKitService.instance.generateToken(
        roomName: roomName,
        userIdentity: userId.toString(),
        userName: (user.username ?? user.fullname ?? '').toString(),
      );

      if (tokenResp == null || tokenResp.token.trim().isEmpty) {
        showSnackBar('Failed to generate LiveKit token');
        return;
      }

      final url =
          tokenResp.livekitUrl.isNotEmpty ? tokenResp.livekitUrl : wsUrl;
      if (isHost) {
        Get.off(() => LiveKitHostScreen(
              livestream: liveData.value,
              url: url,
              token: tokenResp.token,
            ));
      } else {
        Get.off(() => LiveKitAudienceScreen(
              livestream: liveData.value,
              url: url,
              token: tokenResp.token,
            ));
      }
    } catch (e) {
      Loggers.error('[LIVEKIT] start failed: $e');
      showSnackBar('LiveKit error');
    } finally {
      stopLoader();
    }
  }

  Future<void> _bootstrapProviderAndInit() async {
    if (liveData.value.isDummyLive == 1) {
      initVideoPlayer();
      listenLiveStreamData();
      listenUserState();
      fetchLiveStreamComments();
      WakelockPlus.enable();
      return;
    }

    var provider =
        await ConfigService.instance.getLiveProvider(forceRefresh: true);
    if (provider.isEmpty) {
      provider = (setting?.liveProvider ?? '').trim().toLowerCase();
    }
    if (provider.isEmpty) {
      provider = 'livekit';
    }
    if (provider == 'livekit') {
      await _startLiveKit();
      return;
    }

    // If backend returns an unexpected provider, fallback to LiveKit.
    await _startLiveKit();
  }

  bool get isAudioRoom =>
      (liveData.value.liveLayoutMode.toLowerCase() == 'audio');

  bool isUserSeated(int userId) {
    final map = liveData.value.audioSeatMap;
    for (final v in map.values) {
      if (v is int && v == userId) return true;
      final parsed = int.tryParse('$v');
      if (parsed != null && parsed == userId) return true;
    }
    return false;
  }

  int? findAudioSeatIndexForUser(int userId) {
    final map = liveData.value.audioSeatMap;
    for (final entry in map.entries) {
      final v = entry.value;
      final parsed = (v is int) ? v : int.tryParse('$v');
      if (parsed == userId) {
        return int.tryParse(entry.key);
      }
    }
    return null;
  }

  int? _findFirstEmptySeatIndexForUser(int userId) {
    final count = liveData.value.audioSeatCount.clamp(1, 10);
    final currentMap = Map<String, dynamic>.from(liveData.value.audioSeatMap);

    final targetState =
        liveUsersStates.firstWhereOrNull((e) => e.userId == userId);
    final isCohost = targetState?.type == LivestreamUserType.coHost;

    for (int i = 0; i < count; i++) {
      if (isCohost == true && i == 0) continue; // cohost cannot take seat #1
      final v = currentMap['$i'];
      if (v == lockedSeatValue) continue;
      final parsed = (v is int) ? v : int.tryParse('$v');
      final isOccupied = parsed != null && parsed > 0;
      if (!isOccupied) return i;
    }
    return null;
  }

  Future<void> toggleSeatForUser(int userId) async {
    if (!isAudioRoom) return;

    // Normal audience should not auto-sit (even for self). They must request.
    if (userId == myUserId) {
      final myState =
          liveUsersStates.firstWhereOrNull((e) => e.userId == myUserId);
      final isPrivileged = isHost || myState?.type == LivestreamUserType.coHost;
      if (!isPrivileged) {
        onVideoRequestSend(liveData.value);
        return;
      }
    }

    final currentSeatIndex = findAudioSeatIndexForUser(userId);
    if (currentSeatIndex != null) {
      await clearAudioSeat(currentSeatIndex);
      return;
    }

    final seatIndex = _findFirstEmptySeatIndexForUser(userId);
    if (seatIndex == null) {
      showSnackBar('Koi seat empty nahi hai');
      return;
    }

    await assignAudioSeat(seatIndex: seatIndex, userId: userId);
  }

  Future<void> applyLocalAudioGate() async {
    try {
      final forcePublishForMusic = _useNativeLiveMusicMixing &&
          isHost &&
          isLiveMusicEnabled &&
          isLiveMusicPlaying;

      final state =
          liveUsersStates.firstWhereOrNull((e) => e.userId == myUserId);
      if (state == null) {
        if (forcePublishForMusic) {
          await livekitSetMicrophoneEnabled?.call(true);
          final now = DateTime.now().millisecondsSinceEpoch;
          if (now - _lastAudioGateLogMs > 2000) {
            _lastAudioGateLogMs = now;
            Loggers.info(
                '[LIVEKIT] audioGate fallback: state=null forcePublish=$forcePublishForMusic');
          }
        }
        return;
      }

      final isMicOn = state.audioStatus == VideoAudioStatus.on;

      // Video room: only host/cohost can publish mic (prevents audience voice leak).
      // Audio room: anyone seated can publish mic (host/cohost/audience), obeying audioStatus.
      final isHostOrCohost = state.type == LivestreamUserType.host ||
          state.type == LivestreamUserType.coHost;

      // If host is playing live music (native mixing), keep mic published so frames are sent.
      final shouldPublish = forcePublishForMusic
          ? true
          : (!isAudioRoom
              ? (isHostOrCohost && isMicOn)
              : (isUserSeated(myUserId) && isMicOn));

      final now = DateTime.now().millisecondsSinceEpoch;
      if (_lastLocalMicPublish != shouldPublish ||
          now - _lastAudioGateLogMs > 3000) {
        _lastLocalMicPublish = shouldPublish;
        _lastAudioGateLogMs = now;
        Loggers.info(
            '[LIVEKIT] audioGate publish=$shouldPublish micOn=$isMicOn forceMusic=$forcePublishForMusic audioRoom=$isAudioRoom seated=${isUserSeated(myUserId)}');
      }

      // If we are force-publishing for music while user mic is OFF, silence mic contribution.
      if (_useNativeLiveMusicMixing && isHost && isLiveMusicEnabled) {
        final micVol = isMicOn ? 1.0 : 0.0;
        try {
          await webrtc_audio.NativeAudioManagement.liveMusicMixSetMicVolume(
              micVol);
        } catch (_) {}
      }

      await livekitSetMicrophoneEnabled?.call(shouldPublish);
    } catch (_) {}
  }

  Future<void> _applyLocalAudioGateForStatus(VideoAudioStatus status) async {
    try {
      final isMicOn = status == VideoAudioStatus.on;
      final forcePublishForMusic = _useNativeLiveMusicMixing &&
          isHost &&
          isLiveMusicEnabled &&
          isLiveMusicPlaying;

      final state =
          liveUsersStates.firstWhereOrNull((e) => e.userId == myUserId);
      if (state == null) {
        if (forcePublishForMusic) {
          await livekitSetMicrophoneEnabled?.call(true);
        }
        return;
      }

      final isHostOrCohost = state.type == LivestreamUserType.host ||
          state.type == LivestreamUserType.coHost;

      final shouldPublish = forcePublishForMusic
          ? true
          : (!isAudioRoom
              ? (isHostOrCohost && isMicOn)
              : (isUserSeated(myUserId) && isMicOn));

      if (_useNativeLiveMusicMixing && isHost && isLiveMusicEnabled) {
        final micVol = isMicOn ? 1.0 : 0.0;
        try {
          await webrtc_audio.NativeAudioManagement.liveMusicMixSetMicVolume(
              micVol);
        } catch (_) {}
      }

      await livekitSetMicrophoneEnabled?.call(shouldPublish);
    } catch (_) {}
  }

  void setAudioSeatCount(int count) {
    final safe = count.clamp(1, 10);
    final currentMap = Map<String, dynamic>.from(liveData.value.audioSeatMap);

    // Remove assignments outside new range
    currentMap.removeWhere((k, v) {
      final idx = int.tryParse(k) ?? -1;
      return idx < 0 || idx >= safe;
    });

    updateLiveStreamData(audioSeatCount: safe, audioSeatMap: currentMap);
  }

  Future<void> assignAudioSeat({
    required int seatIndex,
    required int userId,
  }) async {
    final count = liveData.value.audioSeatCount.clamp(1, 10);
    if (seatIndex < 0 || seatIndex >= count) return;

    // Co-host cannot take seat NO.1 (index 0)
    final myState = liveUsersStates.firstWhereOrNull((e) => e.userId == userId);
    final isCohost = myState?.type == LivestreamUserType.coHost;
    if (isCohost && seatIndex == 0) {
      showSnackBar('Co-host seat NO.1 par nahi baith sakta');
      return;
    }

    final currentMap = Map<String, dynamic>.from(liveData.value.audioSeatMap);

    // Don't allow assignment on locked seat
    if (currentMap['$seatIndex'] == lockedSeatValue) {
      showSnackBar('Seat locked hai');
      return;
    }

    // Ensure user occupies only one seat
    currentMap.removeWhere((k, v) => (v is int) && v == userId);

    // Assign
    currentMap['$seatIndex'] = userId;

    await updateLiveStreamData(audioSeatMap: currentMap);

    if (userId == myUserId) {
      // optimistic mic gating without waiting for snapshot
      applyLocalAudioGate();
    }
  }

  Future<void> lockAudioSeat(int seatIndex) async {
    final myState =
        liveUsersStates.firstWhereOrNull((e) => e.userId == myUserId);
    final isPrivileged = isHost || myState?.type == LivestreamUserType.coHost;
    if (!isPrivileged) return;
    final count = liveData.value.audioSeatCount.clamp(1, 10);
    if (seatIndex < 0 || seatIndex >= count) return;

    final currentMap = Map<String, dynamic>.from(liveData.value.audioSeatMap);
    final existing = currentMap['$seatIndex'];
    if (existing is int && existing > 0) {
      showSnackBar('Pehle seat empty karo');
      return;
    }
    currentMap['$seatIndex'] = lockedSeatValue;
    await updateLiveStreamData(audioSeatMap: currentMap);
  }

  Future<void> unlockAudioSeat(int seatIndex) async {
    final myState =
        liveUsersStates.firstWhereOrNull((e) => e.userId == myUserId);
    final isPrivileged = isHost || myState?.type == LivestreamUserType.coHost;
    if (!isPrivileged) return;
    final currentMap = Map<String, dynamic>.from(liveData.value.audioSeatMap);
    if (currentMap['$seatIndex'] == lockedSeatValue) {
      currentMap.remove('$seatIndex');
      await updateLiveStreamData(audioSeatMap: currentMap);
    }
  }

  Future<void> clearAudioSeat(int seatIndex) async {
    final currentMap = Map<String, dynamic>.from(liveData.value.audioSeatMap);
    final previous = currentMap['$seatIndex'];
    currentMap.remove('$seatIndex');
    await updateLiveStreamData(audioSeatMap: currentMap);

    final prevParsed = (previous is int) ? previous : int.tryParse('$previous');
    if (prevParsed != null && prevParsed == myUserId) {
      // seat se utarte hi mic publish off
      applyLocalAudioGate();
    }
  }

  void applyBattleFocus() {
    final opponentId = selectedBattleOpponentId.value;
    if (opponentId <= 0) return;

    // Turn off video for non-opponent cohosts (host keeps control)
    final cohosts = liveData.value.coHostIds ?? const [];
    for (final id in cohosts) {
      if (id == opponentId) continue;
      updateUserStateToFirestore(
        id,
        videoStatus: VideoAudioStatus.offByHost,
      );
    }
  }

  void handleRequestResponse({
    required AppUser? user,
    required bool isRefused,
    LivestreamComment? comment,
  }) {
    final userId = user?.userId;
    if (userId == null) return;

    // Only privileged users can accept/refuse requests.
    final myState =
        liveUsersStates.firstWhereOrNull((e) => e.userId == myUserId);
    final isPrivileged = myState?.type == LivestreamUserType.host ||
        myState?.type == LivestreamUserType.coHost;
    if (!isPrivileged) return;

    // Update user state based on refusal
    // Request is only for seat permission, not for making cohost.
    updateUserStateToFirestore(
      userId,
      type: LivestreamUserType.audience,
    );

    // Audio room: if accepted, move user to a seat (they cannot auto-sit).
    // Seat NO.1 (index 0) stays reserved for host.
    if (!isRefused && isAudioRoom) {
      try {
        final count = liveData.value.audioSeatCount.clamp(1, 10);
        final map = Map<String, dynamic>.from(liveData.value.audioSeatMap);
        int? seatIndex;
        for (int i = 1; i < count; i++) {
          final v = map['$i'];
          if (v == lockedSeatValue) continue;
          final parsed = (v is int) ? v : int.tryParse('$v');
          final occupied = parsed != null && parsed > 0;
          if (!occupied) {
            seatIndex = i;
            break;
          }
        }
        if (seatIndex != null) {
          assignAudioSeat(seatIndex: seatIndex, userId: userId);
        } else {
          showSnackBar('Koi seat empty nahi hai');
        }
      } catch (_) {}
    }

    // Determine the comment to delete
    final commentToDelete = comment ??
        comments.firstWhereOrNull((element) =>
            element.senderId == userId &&
            element.commentType == LivestreamCommentType.request);

    if (commentToDelete != null) {
      liveStreamCommentsRef.doc(commentToDelete.id.toString()).delete();
    }
  }

  Future<void> deleteStreamOnFirebase() async {
    final String? roomId = liveData.value.roomID;

    if (roomId == null) {
      Loggers.error('Room ID is null. Cannot stop live stream.');
      return;
    }

    Loggers.info('Stopping live stream Room : $roomId');

    try {
      // Get all users in the livestream and delete them
      QuerySnapshot usersSnapshot = await liveStreamUserStatesRef.get();

      WriteBatch batch = db.batch();

      for (var doc in usersSnapshot.docs) {
        batch.delete(doc.reference);
      }

      Loggers.info(
          'Deleted ${usersSnapshot.docs.length} livestream users_state.');

      // Get all Comments in the livestream and delete them
      QuerySnapshot commentsSnapshot = await liveStreamCommentsRef.get();

      for (var doc in commentsSnapshot.docs) {
        batch.delete(doc.reference);
      }

      Loggers.info(
          'Deleted ${commentsSnapshot.docs.length} livestream comments.');

      // Delete the main live stream document
      batch.delete(liveStreamDocRef);

      // Commit batch delete
      await batch.commit();
      Loggers.success(
          'livestream , users_states , comments  deleted from Firestore.');
    } catch (e, stackTrace) {
      Loggers.error('Failed to stop live stream: $e');
      Loggers.error('StackTrace: $stackTrace');
    }
  }

  void listenLiveStreamData() {
    if (isCall) return; // Skip timeout/logic for calls
    int likeCount = liveData.value.likeCount ?? 0;
    // liveStreamDocListener?.cancel();
    liveStreamDocListener = liveStreamDocRef.snapshots().listen(
      (event) async {
        if (!event.exists) {
          // Host pressed Stop Live (or livestream deleted). Everyone must leave.
          await _forceExitBecauseLiveStopped();
          return;
        }

        final data = event.data() as Map<String, dynamic>?;
        if (data == null) return;
        final stream = Livestream.fromJson(data);

        if (stream.battleType == BattleType.initiate) {
          selectedBattleOpponentId.value = 0;
          timer?.cancel();
          remainingBattleSeconds.value =
              Duration(minutes: stream.battleDuration).inSeconds;
          countdownPlayer.pause();
        }

        if (stream.battleType == BattleType.waiting) {
          totalBattleSecond =
              Duration(minutes: stream.battleDuration).inSeconds;
        }

        // Sync battle opponent from Firestore
        if ((stream.battleOpponentId) > 0) {
          selectedBattleOpponentId.value = stream.battleOpponentId;
        }

        // Update LiveData
        liveData.value = stream;

        await _syncLiveMusicFromSnapshotData(data);

        // Apply focus during PK/battle (host side)
        if (isHost &&
            (stream.battleType == BattleType.waiting ||
                stream.battleType == BattleType.running) &&
            selectedBattleOpponentId.value > 0) {
          applyBattleFocus();
        }

        // Seat map changes should immediately gate mic publish for audio rooms.
        applyLocalAudioGate();

        // Also gate remote play streams in audio rooms (seat + mic status).
        await _syncRemoteAudioGates();

        // If host is in audio mode, ensure host occupies seat NO.1
        ensureHostDefaultSeat();

        // Trigger like animation if changed
        final newLikeCount = stream.likeCount ?? 0;
        if (likeCount != newLikeCount) {
          onLikeTap?.call();
          likeCount = newLikeCount;
        }
      },
      onError: (error) =>
          Loggers.error('Error listening to livestream: $error'),
    );
  }

  Future<void> clearSeatForUser(int userId) async {
    if (!isAudioRoom) return;
    final count = liveData.value.audioSeatCount.clamp(1, 10);
    final currentMap = Map<String, dynamic>.from(liveData.value.audioSeatMap);
    bool changed = false;
    for (int i = 0; i < count; i++) {
      final v = currentMap['$i'];
      final parsed = (v is int) ? v : int.tryParse('$v');
      if (parsed != null && parsed == userId) {
        currentMap.remove('$i');
        changed = true;
      }
    }
    if (changed) {
      await updateLiveStreamData(audioSeatMap: currentMap);
      if (userId == myUserId) {
        applyLocalAudioGate();
      }
    }
  }

  Future<void> clearSeatForUserLatest(int userId) async {
    try {
      final snap = await liveStreamDocRef.get();
      if (!snap.exists) return;
      final data = snap.data() as Map<String, dynamic>?;
      if (data == null) return;

      final mode =
          (data[FirebaseConst.liveLayoutMode] ?? liveData.value.liveLayoutMode)
              .toString()
              .trim()
              .toLowerCase();
      if (mode != 'audio') return;

      final seatCountRaw = data[FirebaseConst.audioSeatCount];
      final seatCount = (seatCountRaw is int)
          ? seatCountRaw
          : int.tryParse('$seatCountRaw') ?? liveData.value.audioSeatCount;
      final count = seatCount.clamp(1, 10);

      final rawMap = data[FirebaseConst.audioSeatMap];
      final currentMap = (rawMap is Map)
          ? Map<String, dynamic>.from(rawMap)
          : Map<String, dynamic>.from(liveData.value.audioSeatMap);

      bool changed = false;
      for (int i = 0; i < count; i++) {
        final v = currentMap['$i'];
        final parsed = (v is int) ? v : int.tryParse('$v');
        if (parsed != null && parsed == userId) {
          currentMap.remove('$i');
          changed = true;
        }
      }
      if (changed) {
        await updateLiveStreamData(audioSeatMap: currentMap);
        if (userId == myUserId) {
          applyLocalAudioGate();
        }
      }
    } catch (_) {}
  }

  void ensureHostDefaultSeat() {
    if (_isExiting) return;
    if (!isHost) return;
    if (!isAudioRoom) return;
    final currentMap = Map<String, dynamic>.from(liveData.value.audioSeatMap);
    if (currentMap['0'] == lockedSeatValue) return;
    final seat0 = currentMap['0'];
    final seat0Id = (seat0 is int) ? seat0 : int.tryParse('$seat0');
    if (seat0Id == myUserId) return;
    // If host already seated elsewhere, do nothing.
    if (isUserSeated(myUserId)) return;
    // If seat0 occupied by someone else, do not override.
    if (seat0Id != null && seat0Id > 0) return;
    // Assign host to seat 0
    assignAudioSeat(seatIndex: 0, userId: myUserId);
  }

  void listenUserState() {
    // Cancel any existing listener
    liveStreamUserStatesListener?.cancel();
    // Loggers.info('👂 Listening to live stream users state...');

    liveStreamUserStatesListener = liveStreamUserStatesRef
        .withConverter(
          fromFirestore: (snapshot, options) {
            if (!snapshot.exists) return null;
            return LivestreamUserState.fromJson(snapshot.data()!);
          },
          toFirestore: (value, options) {
            if (value == null) return {};
            return value.toJson();
          },
        )
        .snapshots()
        .listen((event) {
          // Loggers.info(
          //     '📦 Firestore snapshot received with ${event.docChanges.length} changes');

          for (var change in event.docChanges) {
            final state = change.doc.data();
            if (state == null) {
              // Loggers.warning('⚠️ Null state found in change, skipping...');
              continue;
            }

            switch (change.type) {
              case DocumentChangeType.added:
                _showJoinStreamSheet(state);
                liveUsersStates.add(state);
                // Apply remote mute based on state so new joins respect mic rules.
                _applyRemoteAudioGateForUser(state.userId, state.audioStatus);
                // Loggers.info('➕ User added: ${state.userId}');
                break;

              case DocumentChangeType.modified:
                LivestreamUserState? oldState =
                    liveUsersStates.firstWhereOrNull(
                        (element) => element.userId == state.userId);

                updateStateAction(oldState, state);

                // If someone toggled mic, ensure all listeners hear the change.
                if (oldState == null ||
                    oldState.audioStatus != state.audioStatus) {
                  _applyRemoteAudioGateForUser(state.userId, state.audioStatus);
                }

                int index =
                    liveUsersStates.indexWhere((u) => u.userId == state.userId);
                if (index != -1) {
                  liveUsersStates[index] = state;
                } else {
                  liveUsersStates.add(state);
                }
                // Loggers.info('🔁 User modified: ${state.userId}');
                break;

              case DocumentChangeType.removed:
                liveUsersStates.removeWhere((u) => u.userId == state.userId);
                // If user left, ensure their stream isn't kept muted for next reuse.
                _applyRemoteAudioGateForUser(state.userId, VideoAudioStatus.on);
                // Loggers.info('➖ User removed: ${state.userId}');
                break;
            }
          }
          requestList.value = liveUsersStates
              .where((element) => element.type == LivestreamUserType.requested)
              .toList();
          audienceList.value = liveUsersStates
              .where((element) =>
                  element.type != LivestreamUserType.host &&
                  element.type != LivestreamUserType.left)
              .toList();
          invitedList.value = liveUsersStates
              .where((element) => element.type == LivestreamUserType.invited)
              .toList();
          coHostList.value = liveUsersStates
              .where((element) => element.type == LivestreamUserType.coHost)
              .toList();
          audienceMemberList.value = liveUsersStates
              .where((element) =>
                  element.type != LivestreamUserType.left &&
                  element.userId != myUserId)
              .toList();

          // audioStatus changes should immediately reflect in publish state
          applyLocalAudioGate();
        }, onError: (error) {
          if (error is FirebaseException && error.code == 'permission-denied') {
            Loggers.error(
                'Firestore permission denied (live user state): $error');
            liveStreamUserStatesListener?.cancel();
            return;
          }
          Loggers.error('Firestore live user state listener error: $error');
        });
  }

  void fetchLiveStreamComments() {
    Loggers.info('Fetching live stream comments...');
    liveStreamCommentsListener?.cancel();
    liveStreamCommentsListener = liveStreamCommentsRef
        .withConverter(
          fromFirestore: (snapshot, options) {
            if (!snapshot.exists) {
              Loggers.error('No comments found in Firestore.');
              return null;
            }
            return LivestreamComment.fromJson(snapshot.data()!);
          },
          toFirestore: (value, options) => value?.toJson() ?? {},
        )
        .snapshots()
        .listen((querySnapshot) {
      // Loggers.info(
      //     'Received comment changes: ${querySnapshot.docChanges.length}');

      for (var change in querySnapshot.docChanges) {
        LivestreamComment? comment = change.doc.data();
        if (comment == null) {
          Loggers.error('Null comment received.');
          continue;
        }

        switch (change.type) {
          case DocumentChangeType.added:
            if (comment.commentType == LivestreamCommentType.request &&
                !isHost) {
              continue;
            }
            comments.add(comment);
            // Loggers.info('New comment added: ${comment.toJson()}');
            break;

          case DocumentChangeType.modified:
            if (comment.commentType == LivestreamCommentType.request &&
                !isHost) {
              return;
            }
            int index = comments.indexWhere((c) => c.id == comment.id);
            if (index != -1) {
              comments[index] = comment;
              // Loggers.info('Comment modified: ${comment.toJson()}');
            }
            break;

          case DocumentChangeType.removed:
            comments.removeWhere((c) => c.id == comment.id);
            // Loggers.info('Comment removed: ${comment.id}');
            break;
        }
      }

      // Assign sender and receiver users to comments
      for (var comment in comments) {
        comment.gift =
            gifts.firstWhereOrNull((gift) => gift.id == comment.giftId);
      }

      comments.sort((a, b) => (b.id ?? 0).compareTo(a.id ?? 0));
    }, onError: (error) {
      if (error is FirebaseException && error.code == 'permission-denied') {
        Loggers.error('Firestore permission denied (live comments): $error');
        liveStreamCommentsListener?.cancel();
        return;
      }
      Loggers.error('Firestore live comments listener error: $error');
    });
  }

    if (state?.audioStatus == VideoAudioStatus.offByHost) {
      return showSnackBar(LKey.theHostHasTurnedOffYourAudio);

    }
  }

        liveUsersStates.firstWhereOrNull((e) => e.userId == myUserId);
    final isModerator = isHost || myState?.type == LivestreamUserType.coHost;
    if (!isModerator) return;

    // Never allow toggling host mic via moderator button.

    // Only meaningful for seated users.
    if (!isUserSeated(state.userId)) return;

    // If user turned off their mic themselves, host/cohost shouldn't override.
    if (state.audioStatus == VideoAudioStatus.offByMe) {
      return showSnackBar(LKey.theCoHostHasTurnedOffTheirAudio);
    }

    updateUserStateToFirestore(
      state.userId,
      audioStatus: state.audioStatus == VideoAudioStatus.on
          ? VideoAudioStatus.offByHost
          : VideoAudioStatus.on,
    );
  }

  void insertMention(String username) {
    final safe = username.trim();
    if (safe.isEmpty) return;
    textCommentFocusNode.requestFocus();
    final current = textCommentController.text;
    final cursor = textCommentController.selection.baseOffset;
    final mentionText = '@$safe ';

    if (cursor >= 0 && cursor <= current.length) {
      final before = current.substring(0, cursor);
      final after = current.substring(cursor);
      textCommentController.text = '$before$mentionText$after';
      textCommentController.selection = TextSelection.fromPosition(
        TextPosition(offset: (before + mentionText).length),
      );
    } else {
      textCommentController.text = '$current$mentionText';
      textCommentController.selection = TextSelection.fromPosition(
        TextPosition(offset: textCommentController.text.length),
      );
    }
    isTextEmpty.value = textCommentController.text.trim().isEmpty;
  }

  void onGiftTap(GiftType type,
      {BattleView battleViewType = BattleView.red,
      List<AppUser> users = const []}) {
    // Allow self-gifting in livestream (host/cohost can gift themselves too)
    if (liveData.value.type == LivestreamType.battle &&
        liveData.value.battleType == BattleType.end) {
      return showSnackBar(LKey.battleEndedGiftNotSent.tr);
    }
    GiftManager.openGiftSheet(
        onCompletion: (giftManager) {
          Gift gift = giftManager.gift;
          AppUser? user = giftManager.streamUser;
          final receivers = (giftManager.streamUsers != null &&
                  giftManager.streamUsers!.isNotEmpty)
              ? giftManager.streamUsers!
              : (user == null ? const <AppUser>[] : <AppUser>[user]);

          int coinPrice = gift.coinPrice?.toInt() ?? 0;

          for (final r in receivers) {
            _sendCommentToFirestore(
                type: LivestreamCommentType.gift,
                giftId: gift.id,
                receiverId: r.userId);
            updateUserStateToFirestore(
              r.userId,
              battleCoin: type == GiftType.battle ? coinPrice : null,
              currentBattleCoin: type == GiftType.battle ? coinPrice : null,
              liveCoin: type == GiftType.livestream ? coinPrice : null,
            );
          }
        },
        giftType: type,
        battleViewType: battleViewType,
        streamUsers: users.isEmpty ? getLiveGiftReceivers() : users);
  }

  List<AppUser> getLiveGiftReceivers() {
    final all = <AppUser>[];
    final seen = <int>{};

    void addUserId(int? uid) {
      if (uid == null || uid <= 0) return;
      if (!seen.add(uid)) return;
      final u = firestoreController.users
          .firstWhereOrNull((element) => element.userId == uid);
      if (u != null) all.add(u);
    }

    // Always include host
    addUserId(liveData.value.hostId);

    // Include co-hosts
    final cohostIds = liveData.value.coHostIds ?? const <int>[];
    for (final id in cohostIds) {
      addUserId(id);
    }

    // For audio rooms, include seated occupants
    if (isAudioRoom) {
      final map = liveData.value.audioSeatMap;
      for (final entry in map.entries) {
        final v = entry.value;
        int? uid;
        if (v is int) {
          uid = v;
        } else {
          uid = int.tryParse('$v');
        }
        addUserId(uid);
      }
    }

    // Fallback: if still empty, return whatever we have in firestore list
    if (all.isEmpty) {
      return firestoreController.users;
    }
    return all;
  }

  Future<void> updateUserStateToFirestore(
    int? userId, {
    LivestreamUserType? type,
    VideoAudioStatus? audioStatus,
    VideoAudioStatus? videoStatus,
    int? battleCoin,
    int? liveCoin,
    int? credits,
    bool? isFollow,
    int? joinTime,
    int? currentBattleCoin,
  }) async {
    if (userId == null) {
      Loggers.error('updateUserStateToFirestore: userId is null');
      return;
    }

    DocumentReference reference =
        liveStreamUserStatesRef.doc(userId.toString());
    bool isExist = (await reference.get()).exists;

    if (!isExist) {
      try {
        await reference.set({}, SetOptions(merge: true));
      } catch (e) {
        Loggers.error('updateUserStateToFirestore create failed $userId: $e');
        return;
      }
    }

    try {
      final updateData = <String, dynamic>{
        if (type != null) FirebaseConst.type: type.value,
        if (audioStatus != null) FirebaseConst.audioStatus: audioStatus.value,
        if (videoStatus != null) FirebaseConst.videoStatus: videoStatus.value,
        if (battleCoin != null)
          FirebaseConst.totalBattleCoin:
              battleCoin == 0 ? 0 : FieldValue.increment(battleCoin),
        if (currentBattleCoin != null)
          FirebaseConst.currentBattleCoin: currentBattleCoin == 0
              ? 0
              : FieldValue.increment(currentBattleCoin),
        if (liveCoin != null)
          FirebaseConst.liveCoin:
              liveCoin == 0 ? 0 : FieldValue.increment(liveCoin),
        if (credits != null)
          FirebaseConst.credits:
              credits == 0 ? 0 : FieldValue.increment(credits),
        if (isFollow != null)
          FirebaseConst.followersGained: isFollow
              ? FieldValue.arrayUnion([myUserId])
              : FieldValue.arrayRemove([myUserId]),
        if (joinTime != null) FirebaseConst.joinStreamTime: joinTime,
      };
      if (battleCoin != null || liveCoin != null || credits != null) {
        myUser.value?.coinEstimatedValue(
            battleCoin?.toDouble() ?? liveCoin?.toDouble());
        SessionManager.instance.setUser(myUser.value);
      }
      await liveStreamUserStatesRef.doc(userId.toString()).update(updateData);
      Loggers.success('User state updated for userId: $userId');
    } catch (e, stack) {
      Loggers.error('Failed to update user state: $e\n$stack');
    }
  }

  void closeCoHostStream(int? streamId) {
    StreamView? view = streamViews
        .firstWhereOrNull((element) => element.streamId == '$streamId');
    if (view != null) {
      livekitSetCameraEnabled?.call(false);
      livekitSetMicrophoneEnabled?.call(false);

      // If co-host was seated in audio room, free the seat immediately.
      if (streamId != null) {
        clearSeatForUser(streamId);
      }

      updateLiveStreamData(coHostId: FieldValue.arrayRemove([streamId]));
      LivestreamComment? comment = comments.firstWhereOrNull((element) =>
          element.senderId == myUserId &&
          element.commentType == LivestreamCommentType.joinedCoHost);
      if (comment != null) {
        liveStreamCommentsRef.doc(comment.id.toString()).delete();
      }
      updateUserStateToFirestore(streamId,
          type: LivestreamUserType.audience,
          audioStatus: VideoAudioStatus.offByMe,
          videoStatus: VideoAudioStatus.offByMe,
          battleCoin: 0,
          currentBattleCoin: 0);
      streamViews.removeWhere((element) => element.streamId == '$streamId');
      streamEnded();
    }
  }

  Future<bool> requestPermission() async {
    Loggers.info("requestPermission...");
    try {
      PermissionStatus microphoneStatus = await Permission.microphone.request();
      if (microphoneStatus != PermissionStatus.granted) {
        Loggers.error('Error: Microphone permission not granted!!!');
        return false;
      }
    } on Exception catch (error) {
      Loggers.error("[ERROR], request microphone permission exception, $error");
    }

    try {
      PermissionStatus cameraStatus = await Permission.camera.request();
      if (cameraStatus != PermissionStatus.granted) {
        Loggers.error('Error: Camera permission not granted!!!');
        return false;
      }
    } on Exception catch (error) {
      Loggers.error("[ERROR], request camera permission exception, $error");
    }

    return true;
  }

  void updateStateAction(
      LivestreamUserState? oldState, LivestreamUserState newState) {
    if (newState.userId == myUserId) {
      Loggers.info('Updating state for userId: ${newState.toJson()}');

      // If kicked/forced LEFT by host/cohost, leave immediately.
      if (!_isExiting && newState.type == LivestreamUserType.left) {
        _forceExitBecauseKicked();
        return;
      }

      if (newState.type == LivestreamUserType.coHost &&
          oldState?.type != LivestreamUserType.coHost) {
        publishCoHostStream(myUserId);
      }

      if (newState.type == LivestreamUserType.audience &&
          oldState?.type == LivestreamUserType.invited &&
          isJoinSheetOpen) {
        Get.back();
      }

      if (newState.type == LivestreamUserType.invited &&
          oldState?.type == LivestreamUserType.audience) {
        _showJoinStreamSheet(newState);
      }
      if (oldState?.type == LivestreamUserType.coHost &&
          newState.type == LivestreamUserType.audience) {
        closeCoHostStream(newState.userId);
      }
    }
  }

  Future<void> coHostDelete(LivestreamUserState state) async {
    if (state.type != LivestreamUserType.coHost) return;
    updateLiveStreamData(coHostId: FieldValue.arrayRemove([state.userId]));
    updateUserStateToFirestore(state.userId, type: LivestreamUserType.audience);
    // If audio room, also free the seat so UI doesn't show them still seated.
    try {
      await clearSeatForUserLatest(state.userId);
    } catch (_) {}
  }

  Future<void> kickOutUser(int userId) async {
    if (userId <= 0) return;
    if (userId == myUserId) return;

    // Never allow kicking host.
    if (userId == (liveData.value.hostId ?? -1)) return;

    try {
      // If user was cohost, remove from coHostIds (affects video sections + UI)
      final targetState =
          liveUsersStates.firstWhereOrNull((e) => e.userId == userId);
      if (targetState?.type == LivestreamUserType.coHost) {
        updateLiveStreamData(coHostId: FieldValue.arrayRemove([userId]));
      }

      // Clear seat if audio room
      await clearSeatForUserLatest(userId);

      // Mark as LEFT so their device exits
      updateUserStateToFirestore(
        userId,
        type: LivestreamUserType.left,
        audioStatus: VideoAudioStatus.offByHost,
        videoStatus: VideoAudioStatus.offByHost,
      );
    } catch (_) {}
  }

  void streamEnded() {
    LivestreamUserState? userState = liveUsersStates
        .firstWhereOrNull((element) => element.userId == myUserId);
    AppUser? user = firestoreController.users
        .firstWhereOrNull((element) => element.userId == myUserId);
    userState?.user = user;
    int viewers = liveUsersStates.length;
    if (isHost) {
      Get.back();
      Get.off(() => LiveStreamEndScreen(
          userState: userState, isHost: isHost, viewers: viewers));
    } else {
      if (userState?.type == LivestreamUserType.coHost) {
        Get.bottomSheet(
                LiveStreamSummary(
                    userState: userState, isHost: isHost, viewers: viewers),
                isScrollControlled: true)
            .then((value) {
          updateUserStateToFirestore(myUserId,
              battleCoin: 0,
              liveCoin: 0,
              currentBattleCoin: 0,
              type: LivestreamUserType.audience);
          if ((liveData.value.roomID ?? '').isEmpty) {
            Get.back();
          }
        });
      }
    }
  }

  void battleRunning() {
    Livestream stream = liveData.value;
    // Battle Start Timer Logic

    final startTime =
        DateTime.fromMillisecondsSinceEpoch(stream.battleCreatedAt ?? 0);
    final endTime = startTime
        .add(Duration(seconds: totalBattleSecond + AppRes.battleStartInSecond));

    Loggers.success('Battle Timer Started');

    _timerStart(() {
      final remaining = endTime.difference(DateTime.now()).inSeconds;
      remainingBattleSeconds.value = remaining.clamp(0, totalBattleSecond);

      if (remainingBattleSeconds.value <= 10) {
        if (!countdownPlayer.playing) {
          countdownPlayer
              .seek(Duration(seconds: 10 - remainingBattleSeconds.value));
          countdownPlayer.play();
        }
      }

      Loggers.info(
          '[BATTLE RUNNING] Battle end in ${remainingBattleSeconds.value} sec.');

      if (remainingBattleSeconds.value <= 0) {
        winAudioPlayer.seek(const Duration(seconds: 0));
        winAudioPlayer.play();
        timer?.cancel();
        updateLiveStreamData(battleType: BattleType.end);
      }
    });
  }

  void startMinViewerTimeoutCheck() {
    if (minViewerTimeoutTimer?.isActive ?? false) return;
    Loggers.info(
        'Check Min. Viewers Required to continue live $timeoutMinutes Minutes');
    minViewerTimeoutTimer =
        Timer.periodic(Duration(minutes: timeoutMinutes), (_) {
      minViewerTimeoutTimer?.cancel();
      if ((liveData.value.watchingCount ?? 0) <= minViewersThreshold) {
        isMinViewerTimeout.value = true;
        Loggers.info('Close Stream Because of Min. Viewers');
      }
    });
  }

      backgroundColor: Colors.transparent,
      isDismissible: true,
    );
  }

  void pushNotificationToFollowers(Livestream liveData) {
    AppUser? hostUser = liveData.getHostUser([]);
    NotificationService.instance.pushNotification(
        type: NotificationType.liveStream,
        title: LKey.liveStreamNotificationTitle
            .trParams({'name': hostUser?.username ?? ''}),
        body: LKey.liveStreamNotificationBody.tr,
        deviceType: 1,
        topic: '${liveData.hostId}_ios',
        data: liveData.toJson());
    NotificationService.instance.pushNotification(
        type: NotificationType.liveStream,
        title: LKey.liveStreamNotificationTitle
            .trParams({'name': hostUser?.username ?? ''}),
        body: LKey.liveStreamNotificationBody.tr,
        deviceType: 0,
        topic: '${liveData.hostId}_android',
        data: liveData.toJson());
  }
