///

/// RelativeDistinguishedName ::= SET SIZE (1..MAX) OF AttributeTypeAndValue

  late ASN1Set values;

  ASN1RDN(this.values);

  @override
  Uint8List encode(
      {ASN1EncodingRule encodingRule = ASN1EncodingRule.ENCODING_DER}) {

  }
}
