import 'package:flutter/foundation.dart';
import 'package:flutter/material.dart';

import 'logger.dart';

class ScreenshotManager {
  /// Capture widget screenshot
  static Future<XFile?> captureScreenshot(
    GlobalKey screenshotKey, {
    String? outputFileName,
  }) async {
    try {
      RenderRepaintBoundary? boundary = screenshotKey.currentContext
          ?.findRenderObject() as RenderRepaintBoundary?;
      if (boundary == null) {
        Loggers.info('1');
        return null;
      }

      final image = await boundary.toImage(pixelRatio: 5.0);
      final byteData = await image.toByteData(format: ImageByteFormat.png);
      final Uint8List? imageBytes = byteData?.buffer.asUint8List();
      if (imageBytes == null) {
        Loggers.info('2');
        return null;
      }

      final localPath = await PlatformPathExtension.localPath;

      final name = (outputFileName ?? 'screenshot.png').trim();
      final safeName = name.isEmpty ? 'screenshot.png' : name;
      final file = File('$localPath$safeName');

      await file.writeAsBytes(imageBytes);
      Loggers.info(file.path);
      return XFile(file.path);
    } catch (e) {
      Loggers.error('‚ùå Screenshot failed: $e');
      return null;
    }
  }
}
