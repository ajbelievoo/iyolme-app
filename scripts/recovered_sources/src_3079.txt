  Future<BoostEstimateResponse> getEstimate({
    required int budgetCoins,
    required int durationHours,
    required String placement,
  }) async {
    final response = await ApiService.instance.call(
      url: WebService.boost.estimate,
      fromJson: BoostEstimateResponse.fromJson,
      param: {
        Params.budgetCoins: budgetCoins,
        Params.durationHours: durationHours,
        Params.placement: placement,
      },
    );
    return response;
  }

  Future<List<Post>> fetchBoostedPosts({
    required String placement,
    String? type,
    int? lastItemId,
  }) async {
    final params = <String, dynamic>{
      Params.placement: placement,
      if (type != null) Params.types: type,
      if (lastItemId != null) Params.lastItemId: lastItemId,
    };

    Map<String, dynamic> json;
    try {
      json = await ApiService.instance.call<Map<String, dynamic>>(
        url: WebService.boost.fetchBoostedPosts,
        param: params,
      );
    } catch (_) {
      // Backward-compatible fallback (some backends expose fetch as GET)
      json = await ApiService.instance.get<Map<String, dynamic>>(
        url: WebService.boost.fetchBoostedPosts,
        param: params,
      );
    }

    dynamic raw = json['data'] ??
        json['items'] ??
        json['list'] ??
        json['results'] ??
        json['rows'] ??
        json['posts'] ??
        json['boostedPosts'];

    if (raw is Map) {
      raw = raw['data'] ??
          raw['items'] ??
          raw['list'] ??
          raw['results'] ??
          raw['rows'] ??
          raw['posts'];
    }

    if (raw is List) {
      return raw.map((e) => Post.fromJson(e)).toList();
    }
    return <Post>[];
  }

  Future<StatusModel> trackBoostEvent({
    int? boostId,
    int? postId,
    String? event,
    String? placement,
    String? action,
    String? metadataJson,
  }) async {
    StatusModel model = await ApiService.instance.call(
      url: WebService.boost.trackBoostEvent,
      fromJson: StatusModel.fromJson,

        if (postId != null) Params.postId: postId,
        if (event != null) Params.event: event,
        if (placement != null) Params.placement: placement,
        if (action != null) Params.action: action,
        if (metadataJson != null) Params.metadataJson: metadataJson,
      },
    );
    return model;
  }
