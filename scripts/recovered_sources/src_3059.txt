import 'dart:async';

import 'package:flutter/material.dart';
import 'package:get/get.dart';
import 'package:internet_connection_checker_plus/internet_connection_checker_plus.dart';
import 'package:shortzz/common/manager/logger.dart';
import 'package:shortzz/common/widget/no_internet_sheet.dart';

class InternetConnectionManager {
  InternetConnectionManager._();

  static final instance = InternetConnectionManager._();
  AppLifecycleListener? _appLifecycleListener;

  StreamSubscription<InternetStatus>? listener;
  dynamic retryLastAction;

  void callRetryLastAction([String s = '', dynamic d]) {
    final fn = retryLastAction;
    if (fn == null) return;

    try {
      if (fn is void Function(String, dynamic)) {
        fn(s, d);
        return;
      }
      if (fn is void Function()) {
        fn();
        return;
      }
      if (fn is void Function(dynamic)) {
        fn(d);
        return;
      }
      if (fn is Function) {
        // Best-effort: try common arities to avoid runtime crashes.
        try {
          Function.apply(fn, [s, d]);
          return;
        } catch (_) {}
        try {
          Function.apply(fn, [d]);
          return;
        } catch (_) {}
        try {
          Function.apply(fn, []);
          return;
        } catch (_) {}
      }
    } catch (_) {
      // ignore
    }
  }

  Future<bool> checkInternetConnection() async {
    bool result = await InternetConnection().hasInternetAccess;
    return result;
  }

  void listenNoInternetConnection() {
    listener?.cancel();
    _appLifecycleListener?.dispose();
    listener =
        InternetConnection().onStatusChange.listen((InternetStatus status) {
      Loggers.info("Internet Status : $status");
      switch (status) {
        case InternetStatus.connected:
          if (Get.isOverlaysOpen) {
            try {
              Get.key.currentState?.maybePop();
            } catch (_) {}
          }
          break;
        case InternetStatus.disconnected:
          Get.to(() => const NoInternetSheet(),
              transition: Transition.downToUp);
          break;
      }
    });
    _appLifecycleListener = AppLifecycleListener(
      onResume: () {
        listener?.resume();
        Loggers.success('❤️onResume');
      },
      onHide: () {
        listener?.pause();
        Loggers.success('❤️onHide');
      },
      onPause: () {
        listener?.pause();
        Loggers.success('❤️onPause');
      },
    );
  }

  void cancel() {
    listener?.cancel();
    _appLifecycleListener?.dispose();
  }
}
