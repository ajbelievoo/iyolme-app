enum PostType {
  reel,
  image,
  video,
  text,
  none;

  int get type {
    switch (this) {
      case PostType.reel:
        return 1;
      case PostType.image:
        return 2;
      case PostType.video:
        return 3;
      case PostType.text:
        return 4;
      case PostType.none:
        return 0;
    }
  }

  static String get posts =>
      '${PostType.image.type},${PostType.video.type},${PostType.text.type}';

  static String get reels => '${PostType.reel.type}';

  static PostType fromString(int value) {
    return PostType.values.firstWhere(
      (e) => e.type == value,
      orElse: () => PostType.none,
    );
  }
}

  Future<List<Post>> fetchPostsDiscover({required String type,
      CancelToken? cancelToken,
      int? lastItemId}) async {
    Loggers.info('Fetching posts discover: type=$type, lastItemId=$lastItemId');
    try {
      PostsModel model = await ApiService.instance.call<PostsModel>(
          url: WebService.post.fetchPostsDiscover,
          param: {
            Params.limit: AppRes.paginationLimit,
            Params.types: type,
            Params.lastItemId: lastItemId
          },
          fromJson: (json) => PostsModel.fromJson(json),
          cancelToken: cancelToken);
      
      if (model.data == null || model.data!.isEmpty) {
        Loggers.warning('FetchPostsDiscover returned empty list for type=$type');
      } else {
        Loggers.success('FetchPostsDiscover success: ${model.data!.length} items for type=$type');
      }
      
      return model.data ?? [];
    } catch (e, st) {
      Loggers.error('FetchPostsDiscover failed: $e');
      Loggers.error('FetchPostsDiscover stacktrace: $st');
      rethrow;
    }
  }

  /// Track daily task progress when user performs an action
  void _trackTaskProgress(String taskType) {
    try {
      final controller = Get.isRegistered<ProfessionalController>()
          ? Get.find<ProfessionalController>()
          : null;
      if (controller == null) return;

      // Find matching task and call completeTask API
      final tasks = controller.tasks.where((t) => 
        (t.type?.toLowerCase() == taskType) && (t.completed != true));
      
      for (final task in tasks) {
        if ((task.id ?? 0) > 0) {
          Loggers.info('[TASK_TRACK] Completing task ${task.id} for $taskType');
          // Call completeTask in background - don't await to avoid blocking
          controller.completeTask(task.id!).then((result) {
            if (result.status == true) {
              Loggers.success('[TASK_TRACK] Task ${task.id} completed successfully');
              // Refresh tasks to get updated progress
              controller.refreshTasks();
            }
          }).catchError((e) {
            Loggers.error('[TASK_TRACK] Failed to complete task ${task.id}: $e');
          });
        }
      }
    } catch (e) {
      Loggers.error('[TASK_TRACK] Error tracking task progress: $e');
    }
  }
