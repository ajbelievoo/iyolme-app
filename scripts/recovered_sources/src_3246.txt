  List<AdsManagerCampaignItem> _applyFilters(List<AdsManagerCampaignItem> items) {
    final q = _query.trim().toLowerCase();
    final statusF = _statusFilter.trim().toLowerCase();
    final placementF = _placementFilter.trim().toLowerCase();

    return items.where((c) {
      if (q.isNotEmpty && !c.title.toLowerCase().contains(q)) {
        return false;
      }
      if (statusF != 'all') {
        final s = c.status.toLowerCase();
        if (statusF == 'active') {
          if (!(s == 'active' || s == 'running')) return false;
        } else if (s != statusF) {
          return false;
        }
      }
      if (placementF != 'all') {
        final p = c.placement.toLowerCase();
        if (p != placementF) return false;
      }
      return true;
    }).toList();
  }

  @override
  void initState() {
    super.initState();
    _scrollController.addListener(_onScroll);
    _fetch();
  }

  void _onScroll() {
    if (!_scrollController.hasClients) return;
    if (_loading || _loadingMore || !_hasMore) return;
    final pos = _scrollController.position;
    if (!pos.hasPixels || !pos.hasContentDimensions) return;

    if (pos.pixels >= pos.maxScrollExtent - 260) {
      _fetchMore();
    }
  }

  Future<void> _fetch() async {
    if (!mounted) return;
    setState(() {
      _loading = true;
      _error = null;
      _hasMore = true;
    });

    try {
      final ads = await _repo.listCampaigns();
      final items = <AdsManagerCampaignItem>[];
      items.addAll(ads);
      try {
        final boosts = await _repo.listBoostCampaigns();
        items.addAll(boosts);
      } catch (_) {
        // ignore boost errors, show ads list
      }
      items.sort((a, b) => b.createdAt.compareTo(a.createdAt));
      if (!mounted) return;
      setState(() {
        _items = items;
        _loading = false;
        _hasMore = ads.isNotEmpty;
      });
    } catch (e) {
      if (!mounted) return;
      setState(() {
        _items = null;
        _loading = false;
        _error = e.toString();
        _hasMore = false;
      });
    }
  }

  Future<void> _fetchMore() async {
    final items = _items;
    if (items == null || items.isEmpty) return;
    if (_loadingMore || !_hasMore) return;

    final lastId = items.last.id;
    if (lastId <= 0) return;

    setState(() {
      _loadingMore = true;
    });

    try {
      final next = await _repo.listCampaigns(lastItemId: lastId);
      if (!mounted) return;

      if (next.isEmpty) {
        setState(() {
          _loadingMore = false;
          _hasMore = false;
        });
        return;
      }

      final existingIds = items.map((e) => e.id).toSet();
      final deduped = next.where((e) => !existingIds.contains(e.id)).toList();

      setState(() {
        items.addAll(deduped);
        _loadingMore = false;
        _hasMore = deduped.isNotEmpty;
      });
    } catch (e) {
      if (!mounted) return;
      setState(() {
        _loadingMore = false;
        _hasMore = false;
      });
      BaseController.share.showSnackBar(e.toString());
    }
  }

  Future<void> _toggleCampaign(AdsManagerCampaignItem c, bool nextActive) async {
    final User? user = SessionManager.instance.getUser();
    // final canEdit = user?.hasPermission('campaigns_edit') == true;
    const canEdit = true;
    if (!canEdit) {
      BaseController.share
          .showSnackBar('Forbidden: campaigns_edit permission required');
      return;
    }

    if (_busyCampaignIds.contains(c.id)) return;
    final status = c.status.toLowerCase();
    final prevActive = status == 'active' || status == 'running';
    if (prevActive == nextActive) return;

    final items = _items;
    if (items == null) return;

    final idx = items.indexWhere((x) => x.id == c.id);
    if (idx < 0) return;

    setState(() {
      _busyCampaignIds.add(c.id);
      items[idx] = AdsManagerCampaignItem(
        id: c.id,
        title: c.title,
        mediaUrl: c.mediaUrl,
        status: nextActive ? 'Active' : 'Paused',
        placement: c.placement,
        spend: c.spend,
        budget: c.budget,
        dailyBudget: c.dailyBudget,
        impressions: c.impressions,
        clicks: c.clicks,
        createdAt: c.createdAt,
      );
    });

    try {
      if (nextActive) {
        await _repo.resumeCampaign(campaignId: c.id);
      } else {
        await _repo.pauseCampaign(campaignId: c.id);
      }
    } catch (e) {
      if (!mounted) return;
      setState(() {
        items[idx] = c;
      });
      BaseController.share.showSnackBar(e.toString());
    } finally {
      if (mounted) {
        setState(() {
          _busyCampaignIds.remove(c.id);
        });
      }
    }
  }

  Future<void> _deleteCampaign(AdsManagerCampaignItem c) async {
    final confirm = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('Delete Campaign'),
        content: Text('Are you sure you want to delete "${c.title}"?'),
        actions: [
          TextButton(
            onPressed: () => Navigator.of(ctx).pop(false),
            child: const Text('Cancel'),
          ),
          TextButton(
            onPressed: () => Navigator.of(ctx).pop(true),
            child: const Text('Delete', style: TextStyle(color: Colors.red)),
          ),
        ],
      ),
    );

    if (confirm != true) return;

    final items = _items;
    if (items == null) return;
    
    // Optimistic remove
    setState(() {
      items.removeWhere((x) => x.id == c.id);
    });

    try {
      await _repo.deleteCampaign(campaignId: c.id);
      BaseController.share.showSnackBar('Campaign deleted');
    } catch (e) {
      // Revert
      setState(() {
        items.add(c);
        // Sort or insert at correct position if needed, but append is safe for now
      });
      BaseController.share.showSnackBar(e.toString());
    }
  }

                      onChanged: (!canToggle || busy || !canEdit)
                          ? null
                          : (v) {
                              _toggleCampaign(c, v);
                            },
                    ),
                  ],
                ),
                const SizedBox(height: 4),
                Text(
                  '${c.placement}  â€¢  Status: ${c.status}',
                  style: TextStyle(
                    color: textLightGrey(context),
                    fontSize: 12,
                  ),
                ),
                const SizedBox(height: 10),
                ClipRRect(
                  borderRadius: BorderRadius.circular(999),
                  child: LinearProgressIndicator(
                    value: progress.clamp(0, 1).toDouble(),
                    minHeight: 8,
                    backgroundColor: Colors.black.withValues(alpha: 0.05),
                    valueColor: AlwaysStoppedAnimation<Color>(
                      themeAccentSolid(context),
                    ),
                  ),
                ),
                const SizedBox(height: 8),
                Row(
                  children: [
                    Expanded(
                      child: Text(
                        'Spent: ${c.spend} / ${c.budget}',
                        style: TextStyle(
                          fontSize: 12,
                          color: textLightGrey(context),
                        ),
                      ),
                    ),
                    TextButton(
                      onPressed: (status == 'active' || status == 'paused')
                          ? (canEdit ? () => _openEditBudgetSheet(c) : () {
                              BaseController.share.showSnackBar(
                                'Forbidden: campaigns_edit permission required',
                              );
                            })
                          : null,
                      child: const Text('Edit Budget'),
                    ),
                    TextButton(
                      onPressed: canAnalytics
                          ? () {
                              Navigator.of(context).push(
                                MaterialPageRoute(
                                  builder: (_) => CampaignAnalyticsScreen(
                                    campaign: c,
                                  ),
                                ),
                              );
                            }
                          : () {
                              BaseController.share.showSnackBar(
                                'Forbidden: analytics_access permission required',
                              );
                            },
                      child: const Text('Analytics'),
                    ),
                    if (canEdit)
                      PopupMenuButton<String>(
                        icon: Icon(Icons.more_vert, color: textLightGrey(context)),
                        onSelected: (v) {
                          if (v == 'delete') {
                            _deleteCampaign(c);
                          } else if (v == 'toggle') {
                            _toggleCampaign(c, !isActive);
                          }
                        },
                        itemBuilder: (ctx) => [
                          PopupMenuItem(
                            value: 'toggle',
                            child: Text(isActive ? 'Pause' : 'Activate'),
                          ),
                          const PopupMenuItem(
                            value: 'delete',
                            child: Text('Delete', style: TextStyle(color: Colors.red)),
                          ),
                        ],
                      ),
                  ],
                ),
              ],
            ),
          );
        },
      );
    }

    return RefreshIndicator(
      onRefresh: _fetch,
      child: PrimaryScrollController(
        controller: _scrollController,
        child: body,
      ),
    );
  }

  @override
  void dispose() {
    _scrollController.removeListener(_onScroll);
    _scrollController.dispose();
    super.dispose();
  }
