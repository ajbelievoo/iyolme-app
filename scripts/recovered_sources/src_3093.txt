  Future<void> _syncDeviceTokenAfterLogin() async {
    try {
      final token = await FirebaseMessaging.instance.getToken();
      if (token == null || token.isEmpty) return;
      Loggers.info(
          '[FCM_SYNC_AFTER_LOGIN] tokenLen=${token.length} authTokenEmpty=${(SessionManager.instance.getAuthToken()).contains('EMPTY')}');
      final updated = await updateUserDetails(
        deviceToken: token,
        device: Platform.isAndroid ? 0 : 1,
      );
      if (updated?.deviceToken == token) {
        Loggers.success('âœ… Device token synced after login');
      } else {
        Loggers.error(
            '[FCM_SYNC_AFTER_LOGIN] backend did not echo token. userNull=${updated == null} backendToken=${updated?.deviceToken}');
      }
    } catch (e) {
      Loggers.error('Device token sync after login failed: $e');
    }
  }

  Future<User?> fetchUserDetails({
    int? userId,
    Function()? onError,
    bool forceRefresh = false,
  }) async {
    final uid = userId ?? SessionManager.instance.getUserID();
    if (uid <= 0) {
      try {
        onError?.call();
      } catch (_) {}
      return null;
    }

    final now = DateTime.now().millisecondsSinceEpoch;
    if (!forceRefresh) {
      final cached = _fetchUserDetailsCache[uid];
      if (cached != null) {
        final age = now - cached.fetchedAtMs;
        if (age >= 0 && age < _fetchUserDetailsCacheMs) {
          return cached.user;
        }
      }
    }

    final existingReq = _fetchUserDetailsInFlight[uid];
    if (existingReq != null) {
      return existingReq;
    }

    final req = () async {
      assert(() {
        Loggers.info('[FETCH_USER_DETAILS] request userId=$uid');
        return true;
      }());
      try {
        UserModel userModel = await ApiService.instance.call(
            url: WebService.user.fetchUserDetails,
            param: {Params.userId: uid},
            fromJson: UserModel.fromJson,
            onError: onError);
        assert(() {
          Loggers.info(
              '[FETCH_USER_DETAILS] response userId=$uid status=${userModel.status} isVerify=${userModel.data?.isVerify} username=${userModel.data?.username}');
          return true;
        }());
        final u = userModel.data;
        _fetchUserDetailsCache[uid] =
            _CachedUserDetails(u, DateTime.now().millisecondsSinceEpoch);
        if (userModel.status == true &&
            userId == SessionManager.instance.getUserID()) {
          SessionManager.instance.setUser(u);
        }
        return u;
      } finally {
        _fetchUserDetailsInFlight.remove(uid);
      }
    }();

    _fetchUserDetailsInFlight[uid] = req;
    return req;
  }

            if (notifyMention != null)

            if (notifyGiftReceived != null)
              Params.notifyGiftReceived: notifyGiftReceived ? 1 : 0,

  Future<StatusModel> completeTask({required int taskId}) async {
    final candidatesRaw = <String>[
      WebService.user.completeTask,
      '${apiURL}user/completeTask',
      '${apiURL}user/complete-task',
      '${apiURL}completeTask',
      '${apiURL}complete-task',
      '${baseURL}user/completeTask',
      '${baseURL}api/user/completeTask',
    ];
    
    final seen = <String>{};
    final candidates = <String>[];
    for (final u in candidatesRaw) {
      final url = u.trim();
      if (url.isEmpty || seen.contains(url)) continue;
      seen.add(url);
      candidates.add(url);
    }
    
    Object? lastError;
    String? lastUrl;
    final tried = <String>[];
    
    for (final url in candidates) {
      try {
        lastUrl = url;
        tried.add(url);
        Loggers.info('[COMPLETE_TASK] Trying URL: $url with taskId: $taskId');
        
        final StatusModel response = await ApiService.instance.call(
          url: url,
          param: {
            'task_id': taskId,
            'id': taskId,
          },
          fromJson: StatusModel.fromJson,
        );
        
        Loggers.info('[COMPLETE_TASK] SUCCESS on $url: status=${response.status}');
        return response;
      } catch (e) {
        Loggers.info('[COMPLETE_TASK] ERROR on $url: $e');
        lastError = e;
        // Continue to next candidate
      }
    }
    
    throw Exception(
      'URL Error: tried=${tried.join(', ')} last=${lastUrl ?? ''} - ${lastError?.toString() ?? 'Failed to complete task'}',
    );
  }

  // New: Leaderboards (server returns top 100; client can locally page)
  Future<LeaderboardModel> leaderboard({
    required String type, // 'creator' | 'miner'
    String? metricType, // creator only: engagement|views|likes|comments
    String? period, // daily|weekly|monthly|all
  }) async {
    final bool isCreator = type == 'creator';
    final String url = isCreator
        ? WebService.user.getCreatorLeaderboard
        : WebService.user.getMinerLeaderboard;

    // Backend expects POST. It may return list under different keys.
    final LeaderboardModel response = await ApiService.instance.call(
      url: url,
      param: {
        if (isCreator && (metricType != null)) Params.type: metricType,
        if (period != null) 'period': period,
      },
      fromJson: (json) {
        // Normalize list key
        final dynamic raw = json['data'] ??
            json['leaderboard'] ??
            json['result'] ??
            json['list'];
        final Map<String, dynamic> normalized = {
          'status': json['status'],
          'message': json['message'],
          'data': raw is List
              ? raw
              : (raw is Map ? (raw['data'] ?? raw['list']) : null),
        };
        return LeaderboardModel.fromJson(normalized);
      },
    );
    return response;
  }

  Future<SubscriptionPlansModel> subscriptionPlans() async {

    final candidates = <String>[

    ];

    Object? lastError;

    for (final url in candidates) {

            .call(

              param: <String, dynamic>{},
              fromJson: SubscriptionPlansModel.fromJson,
            )
            .timeout(const Duration(seconds: 12));
        return response;
      } catch (e) {
        lastError = e;
      }
    }

  Future<VerificationSubmitModel> submitVerificationRequestMultipart({
    required String fullName,
    required String documentType,
    required String documentNumber,
    String? address,
    String? reasonForVerification,
    String? accountCategory,
    int? subscriptionId,
    String? subscriptionName,
    String? paymentId,
    XFile? documentFront,
    XFile? documentBack,
    XFile? selfie,
  }) async {
    final current = SessionManager.instance.getUser();
    final username = (current?.username ?? '').trim();
    final mobile = (current?.userMobileNo ?? '').trim();

    Loggers.info(
        '[SUBMIT_VERIFICATION] username=$username mobile=$mobile subscriptionId=$subscriptionId paymentId=$paymentId');

    final param = <String, dynamic>{
      // Backend expects these keys
      if (fullName.trim().isNotEmpty) Params.fullName: fullName,
      if (username.isNotEmpty) Params.username: username,
      if (mobile.isNotEmpty) Params.mobileNumber: mobile,
      Params.documentType: documentType,
      Params.documentNumber: documentNumber,
      if (address != null && address.trim().isNotEmpty)
        Params.address: address.trim(),
      Params.reasonForVerification: reasonForVerification ?? 'subscription',
      // Backend also falls back to `reason`.
      Params.reason: reasonForVerification ?? 'subscription',
      Params.accountCategory: accountCategory ?? 'individual',
      if (subscriptionId != null && subscriptionId > 0)
        Params.subscriptionId: subscriptionId,
      if (subscriptionName != null && subscriptionName.trim().isNotEmpty)
        Params.subscriptionName: subscriptionName.trim(),
      if (paymentId != null && paymentId.trim().isNotEmpty)
        Params.paymentId: paymentId.trim(),
    };

    Loggers.info('[SUBMIT_VERIFICATION] PARAMS: $param');

    final files = <String, List<XFile?>>{};
    final dt = documentType.trim().toLowerCase();
    if (dt == 'aadhaar' || dt == 'aadhar') {
      if (documentFront != null) files[Params.aadhaarFront] = [documentFront];
      if (documentBack != null) files[Params.aadhaarBack] = [documentBack];
    } else if (dt == 'pan') {
      if (documentFront != null) files[Params.panCard] = [documentFront];
    } else if (dt == 'passport') {
      if (documentFront != null) files[Params.passport] = [documentFront];
    } else {
      if (documentFront != null) files[Params.documentFront] = [documentFront];
      if (documentBack != null) files[Params.documentBack] = [documentBack];
    }
    if (selfie != null) {
      files[Params.selfieWithId] = [selfie];
    }

    Loggers.info('[SUBMIT_VERIFICATION] FILES KEYS: ${files.keys.toList()}');

    final candidatesRaw = <String>[
      // Backend route is at API root (no /user)
      WebService.user.submitVerificationRequestV2,
      WebService.user.submitVerificationRequest,
      '${apiURL}submitVerificationRequest',
      '${apiURL}submitVerificationRequest/',
      // Some deployments might not have the /api prefix
      '${baseURL}submitVerificationRequest',
      '${baseURL}submitVerificationRequest/',
    ];
    final seen = <String>{};
    final candidates = <String>[];
    for (final u in candidatesRaw) {
      final url = u.trim();
      if (url.isEmpty || seen.contains(url)) continue;
      seen.add(url);
      candidates.add(url);
    }

    Object? lastError;
    String? lastUrl;
    final tried = <String>[];
    for (final url in candidates) {
      try {
        lastUrl = url;
        tried.add(url);
        Loggers.info('[SUBMIT_VERIFICATION] TRYING URL: $url');
        final VerificationSubmitModel response = await ApiService.instance
            .multiPartCallApi(
              url: url,
              param: Map<String, dynamic>.from(param),
              filesMap: files,
              fromJson: VerificationSubmitModel.fromJson,
            )
            .timeout(const Duration(seconds: 35));
        Loggers.success(
            '[SUBMIT_VERIFICATION] SUCCESS status=${response.status} message=${response.message}');
        return response;
      } catch (e) {
        lastError = e;
        Loggers.error('[SUBMIT_VERIFICATION] ERROR on $url: $e');
      }
    }

    throw Exception(
      'URL Error: tried=${tried.join(', ')} last=${lastUrl ?? ''} - ${lastError?.toString() ?? 'Failed to submit verification request'}',
    );
  }

  Future<VerificationStatusModel> fetchVerificationStatus({
    int? requestId,
  }) async {
    final candidatesRaw = <String>[
      WebService.user.getVerificationStatus,
      '${apiURL}getVerificationStatus',
      '${apiURL}getVerificationStatus/',
      WebService.user.verificationStatus,
      '${apiURL}verification-status',
      '${apiURL}verification-status/',
    ];
    final seen = <String>{};
    final candidates = <String>[];
    for (final u in candidatesRaw) {
      final url = u.trim();
      if (url.isEmpty || seen.contains(url)) continue;
      seen.add(url);
      candidates.add(url);
    }

    final param = <String, dynamic>{
      Params.userId: SessionManager.instance.getUserID(),
      if (requestId != null && requestId > 0)
        Params.verificationRequestId: requestId,
    };

    Object? lastError;
    String? lastUrl;
    String? lastMethod;
    for (final url in candidates) {
      // Try GET first.
      try {
        lastUrl = url;
        lastMethod = 'GET';
        final VerificationStatusModel response = await ApiService.instance.get(
          url: url,
          param: Map<String, dynamic>.from(param),
          fromJson: VerificationStatusModel.fromJson,
        );
        return response;
      } catch (e) {
        lastError = e;
      }

      // Many backends use POST.
      try {
        lastUrl = url;
        lastMethod = 'POST';
        final VerificationStatusModel response = await ApiService.instance.call(
          url: url,
          param: Map<String, dynamic>.from(param),
          fromJson: VerificationStatusModel.fromJson,
        );
        return response;
      } catch (e) {
        lastError = e;
      }
    }

    throw Exception(
      'URL Error: ${lastUrl ?? ''} method=${lastMethod ?? ''} - ${lastError?.toString() ?? 'Failed to fetch verification status'}',
    );
  }
